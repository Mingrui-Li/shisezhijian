import {
  _decorator, BlockInputEvents, Color, Component, EventTouch, Graphics,
  JsonAsset, Label, Layers, Node, ResolutionPolicy, resources, sys, tween,
  UITransform, view, game as engineGame, Game, Sprite, SpriteFrame, Vec3, Rect, Size,
} from 'cc';
import { Action, adaptLevel, createGame, GameState, LevelDefinition, reduceGame, startLevel, recoverInterruptedGame, PROP_PRICES, canClaimDoubleScore } from './core';
import { completeLevel, emptyProgress, ProgressSave, unlockedThrough } from './application/Progress';
import { LocalSave } from './platform/LocalSave';
import { RewardedVideo } from './platform/RewardedVideo';
import { theme } from './theme/Theme';
import { boardLayout } from './render/BoardLayout';
import { Feedback } from './platform/Feedback';
import { BrowserTestBridge } from './platform/BrowserTestBridge';
import { runtimeConfig } from './application/RuntimeConfig';
import { LoadingScreen } from './render/LoadingScreen';
import { TutorialScreen } from './render/TutorialScreen';
import { WinPanel } from './render/WinPanel';

const { ccclass } = _decorator;
type Prop = 'locate' | 'peek' | 'addTime';
const C = theme.colors;

/** One scene, engine-native UI, and a pure state reducer: no DOM / web framework. */
@ccclass('ShiseBoot')
export class ShiseBoot extends Component {
  private root!: Node;
  private height = 844;
  private top = 24;
  private bottom = 20;
  private levels: LevelDefinition[] = [];
  private save: ProgressSave = emptyProgress('');
  private storage = new LocalSave();
  private video = new RewardedVideo({ adUnitId: runtimeConfig.rewardedAdUnitId,
    runtime: (globalThis as unknown as { tt?: object }).tt || {} });
  private feedback!: Feedback;
  private bridge!: BrowserTestBridge;
  private homeFrame: SpriteFrame | null = null;
  private splashFrame: SpriteFrame | null = null;
  private iconsFrame: SpriteFrame | null = null;
  private homeArtFrame: SpriteFrame | null = null;
  private winFrame: SpriteFrame | null = null;
  private iconFrames = new Map<string, SpriteFrame>();
  private loading: LoadingScreen | null = null;
  private tutorial: TutorialScreen | null = null;
  private winPanel: WinPanel | null = null;
  private ready = false;
  private screen: 'home' | 'levels' | 'game' | 'shop' | 'tutorial' = 'home';
  private chapterPage = 0;
  private selectedCell: string | null = null;
  private selectedPool: number | null = null;
  private trayPage = 0;
  private modalOpen = false;
  private notice = '';
  private actionBusy = false;
  private idSequence = 0;
  private dismissedFailure: string | null = null;
  private timerLabel: Label | null = null;
  private clockAt = Date.now();
  private clockSavedAt = 0;
  private appHidden = false;

  onLoad(): void {
    const frame = view.getFrameSize();
    view.setDesignResolutionSize(390, 844, frame.width / frame.height > 0.65
      ? ResolutionPolicy.SHOW_ALL : ResolutionPolicy.FIXED_WIDTH);
    this.height = view.getVisibleSize().height;
    const safe = sys.getSafeAreaRect(false);
    this.top = Math.max(24, this.height - safe.yMax + 12);
    this.bottom = Math.max(20, safe.yMin + 12);
    try {
      const tt = (globalThis as unknown as { tt?: {
        getMenuButtonBoundingClientRect?: () => { bottom: number };
        getSystemInfoSync?: () => { windowWidth: number };
      } }).tt;
      const menu = tt?.getMenuButtonBoundingClientRect?.();
      const width = tt?.getSystemInfoSync?.().windowWidth;
      if (menu && width) this.top = Math.max(this.top, menu.bottom * 390 / width + 10);
    } catch { /* The safe-area fallback remains valid on hosts without a capsule API. */ }
    this.feedback = new Feedback(this.node);
    this.bridge = new BrowserTestBridge(runtimeConfig.browserAccessibility, () => ({
      screen: this.loading ? 'loading' : this.screen, modalOpen: this.modalOpen, version: runtimeConfig.version,
      level: this.save.game?.level.number ?? null, status: this.save.game?.attempt.status ?? null,
      remainingTimeMs: this.save.game?.attempt.remainingTimeMs ?? null, points: this.save.game?.wallet.points ?? 0, trayPage: this.trayPage + 1,
      inventory: this.save.game?.wallet.inventory ?? null, completed: this.save.completed.length,
    }), this.height);
    this.root = new Node('ShiseUI');
    this.root.layer = Layers.Enum.UI_2D;
    this.root.addComponent(UITransform).setContentSize(390, this.height);
    this.node.addChild(this.root);
    engineGame.on(Game.EVENT_HIDE, this.suspend, this);
    engineGame.on(Game.EVENT_SHOW, this.resume, this);
    this.renderLoading();
    // First launch always reaches the home screen. Tutorial is an explicit home entry.
    let assetsDone = 0;
    const art = ['splash-v2', 'home-v2', 'icons-v2', 'win-rosette-v1'];
    const artworkReady = Promise.all(art.map((name, index) => new Promise<void>(resolve => {
      resources.load(`art/${name}/spriteFrame`, SpriteFrame, (error, frame) => {
        if (!error && frame) {
          if (index === 0) { this.splashFrame = frame; this.loading?.setArtwork(frame); }
          if (index === 1) this.homeFrame = frame;
          if (index === 2) this.iconsFrame = frame;
          if (index === 3) this.winFrame = frame;
        } else console.error(`[Shise] Artwork unavailable: ${name}`, error);
        this.loading?.setProgress(0.12 + ++assetsDone * 0.20);
        resolve();
      });
    })));
    const loadingStarted = Date.now();
    resources.load('generated/levels', JsonAsset, async (error, asset) => {
      if (error || !asset) {
        this.clear();
        this.text('颜色还没准备好', 195, this.height / 2 - 24, 280, 30, 22);
        this.text('请检查 generated/levels 资源', 195, this.height / 2 + 16, 300, 40, 14, C.textSecondary);
        return;
      }
      try {
        const content = asset.json as { contentVersion: string; levels: unknown[] };
        this.levels = content.levels.map(adaptLevel);
        this.save = this.storage.load(content.contentVersion, this.levels);
        if (!this.save.game) this.save.game = createGame(this.levels[0], this.nextId('attempt'));
        this.persist();
        // Only valid level versions can be resumed; do not reinterpret old boards.
        if (this.save.game && !this.levels.some(l => l.id === this.save.game!.level.id)) this.save.game = null;
        await artworkReady;
        this.loading?.setProgress(1, '准备好了');
        // Keep the real completion state visible long enough for readable launch notices.
        const finish = () => {
          if (this.loading?.isAgeInfoOpen) { this.scheduleOnce(finish, 0.2); return; }
          this.ready = true; this.renderHome();
        };
        this.scheduleOnce(finish, Math.max(0.25, (3000 - (Date.now() - loadingStarted)) / 1000));
      } catch (error) {
        console.error('[Shise] Failed to adapt level content', error);
        this.clear();
        this.text('题库版本需要更新', 195, this.height / 2, 300, 40, 22);
      }
    });
  }

  onDestroy(): void {
    this.tutorial?.dispose(); this.loading?.destroy(); this.winPanel?.dispose();
    engineGame.off(Game.EVENT_HIDE, this.suspend, this);
    engineGame.off(Game.EVENT_SHOW, this.resume, this);
    this.video.dispose();
  }

  private resume(): void { this.appHidden = false; this.clockAt = Date.now(); if (this.screen === 'game') {
    this.renderGame();
  } }

  private suspend(): void {
    this.consumeTime(true); this.appHidden = true;
    this.selectedCell = null; this.selectedPool = null;
    if (this.save.game?.attempt.status === 'PreviewingReference')
      this.save.game = recoverInterruptedGame(this.save.game);
    this.persist();
  }

  private nextId(prefix: string): string { return `${prefix}-${Date.now()}-${++this.idSequence}`; }

  update(): void {
    const wasPlaying = this.save.game?.attempt.status === 'Playing';
    this.consumeTime();
    if (wasPlaying && this.save.game?.attempt.status === 'Failed' && this.screen === 'game') this.renderGame();
  }

  private consumeTime(force = false): void {
    const now = Date.now();
    if (!this.ready || this.screen !== 'game' || this.modalOpen || this.appHidden || this.save.game?.attempt.status !== 'Playing') {
      this.clockAt = now; return;
    }
    const deltaMs = Math.max(0, now - this.clockAt);
    if (!force && deltaMs < 200) return;
    this.clockAt = now;
    if (!deltaMs) return;
    this.save.game = reduceGame(this.save.game, { type: 'elapse', deltaMs }).state;
    this.updateTimerLabel();
    if (now - this.clockSavedAt >= 5000 || this.save.game.attempt.status === 'Failed') {
      this.persist(); this.clockSavedAt = now;
    }
  }

  private updateTimerLabel(): void {
    if (!this.timerLabel?.isValid || !this.save.game) return;
    const seconds = Math.ceil(this.save.game.attempt.remainingTimeMs / 1000);
    this.timerLabel.string = `${Math.floor(seconds / 60)}:${String(seconds % 60).padStart(2, '0')}`;
    this.timerLabel.color = new Color().fromHEX(seconds <= 10 ? C.warning : C.text);
  }

  private persist(): boolean {
    const stored = this.storage.write(this.save);
    if (!stored || this.storage.error) this.notice = this.storage.error;
    return stored;
  }

  private persistConfirmedReward(retry = 0): void {
    if (this.persist()) return;
    this.notice = '奖励已确认，正在重试保存，请暂时不要关闭游戏';
    if (retry < 3) this.scheduleOnce(() => this.persistConfirmedReward(retry + 1), 2);
    else this.notice = '奖励已确认，但本地保存仍失败，请保留游戏并稍后重试';
  }

  private clear(): void {
    this.timerLabel = null;
    this.tutorial?.dispose(); this.tutorial = null;
    this.winPanel?.dispose(); this.winPanel = null;
    this.loading?.destroy(); this.loading = null;
    this.bridge.clear();
    for (const node of [...this.root.children]) { node.removeFromParent(); node.destroy(); }
    this.modalOpen = false;
    this.round(195, this.height / 2, 392, this.height + 2, 0, C.background);
  }

  private element(name: string, x: number, y: number, width: number, height: number, parent = this.root): Node {
    const node = new Node(name);
    node.layer = Layers.Enum.UI_2D;
    node.addComponent(UITransform).setContentSize(width, height);
    node.setPosition(x - 195, this.height / 2 - y);
    parent.addChild(node);
    return node;
  }

  private round(x: number, y: number, width: number, height: number, radius: number,
    hex: string, parent = this.root, stroke?: string): Node {
    const node = this.element('Shape', x, y, width, height, parent);
    const g = node.addComponent(Graphics);
    g.fillColor = new Color().fromHEX(hex);
    g.roundRect(-width / 2, -height / 2, width, height, Math.min(radius, height / 2, width / 2));
    g.fill();
    if (stroke) { g.strokeColor = new Color().fromHEX(stroke); g.lineWidth = 2; g.stroke(); }
    return node;
  }

  private text(value: string, x: number, y: number, width: number, height: number,
    size = 16, color: string = C.text, bold = false, parent = this.root): Node {
    this.bridge.text(value);
    const node = this.element('Text', x, y, width, height, parent);
    const label = node.addComponent(Label);
    label.string = value;
    label.fontFamily = theme.typography.family;
    label.useSystemFont = true;
    label.fontSize = size;
    label.lineHeight = size * 1.45;
    label.color = new Color().fromHEX(color);
    label.isBold = bold;
    label.horizontalAlign = Label.HorizontalAlign.CENTER;
    label.verticalAlign = Label.VerticalAlign.CENTER;
    label.overflow = Label.Overflow.SHRINK;
    return node;
  }

  private button(value: string, x: number, y: number, width: number, height: number,
    onTap: () => void, fill: string = C.primary, color: string = C.onPrimary,
    parent = this.root, modalButton = false, semanticLabel?: string): Node {
    const node = this.round(x, y, width, height, theme.radii.button, fill, parent);
    // Label and shape share the same center; labels deliberately do not carry input.
    const label = new Node('Label');
    label.layer = Layers.Enum.UI_2D;
    label.addComponent(UITransform).setContentSize(width - 16, height);
    const component = label.addComponent(Label);
    component.string = value;
    component.fontFamily = theme.typography.family;
    component.useSystemFont = true;
    component.fontSize = height < 50 ? 15 : 18;
    component.lineHeight = 26;
    component.isBold = false;
    component.color = new Color().fromHEX(color);
    component.horizontalAlign = Label.HorizontalAlign.CENTER;
    component.verticalAlign = Label.VerticalAlign.CENTER;
    component.overflow = Label.Overflow.SHRINK;
    node.addChild(label);
    node.on(Node.EventType.TOUCH_START, () => {
      if ((!this.modalOpen || modalButton) && !this.save.settings.reducedMotion)
        tween(node).to(0.09, { scale: new Vec3(0.97, 0.97, 1) }).start();
    });
    const release = () => {
      if (this.save.settings.reducedMotion) node.setScale(1, 1, 1);
      else tween(node).to(0.09, { scale: new Vec3(1, 1, 1) }).start();
    };
    const invoke = () => { if ((!this.modalOpen || modalButton) && !this.actionBusy) onTap(); };
    this.bridge.control(semanticLabel || value, invoke, { x, y, width, height });
    node.on(Node.EventType.TOUCH_CANCEL, release);
    node.on(Node.EventType.TOUCH_END, (event: EventTouch) => {
      event.propagationStopped = true;
      release();
      invoke();
    });
    return node;
  }

  private renderLoading(): void {
    this.clear();
    this.loading = new LoadingScreen({ parent: this.root, width: 390, height: this.height,
      safeTop: this.top, safeBottom: this.bottom, artwork: this.splashFrame || undefined,
      registerControl: (label, invoke, bounds) => this.bridge.control(label, invoke, bounds),
      clearControls: labels => this.bridge.removeControls(labels) });
    this.loading.setProgress(0.05, '正在准备颜色');
  }

  private artwork(frame: SpriteFrame, x: number, y: number, width: number, height: number, parent = this.root): Node {
    const node = this.element('Illustration', x, y, width, height, parent);
    const sprite = node.addComponent(Sprite); sprite.sizeMode = Sprite.SizeMode.CUSTOM; sprite.spriteFrame = frame;
    node.getComponent(UITransform)!.setContentSize(width, height);
    return node;
  }

  private icon(name: string, x: number, y: number, size = 40, parent = this.root): void {
    const names = ['home', 'tutorial', 'settings', 'shop', 'locate', 'peek', 'addTime', 'undo', 'back', 'next', 'return', 'cancel'];
    const index = names.indexOf(name);
    if (this.iconsFrame && index >= 0) {
      let frame = this.iconFrames.get(name);
      if (!frame) {
        const source = this.iconsFrame;
        const width = source.originalSize.width / 4, height = source.originalSize.height / 3;
        frame = new SpriteFrame(); frame.texture = source.texture;
        frame.rect = new Rect(index % 4 * width, Math.floor(index / 4) * height, width, height);
        this.iconFrames.set(name, frame);
      }
      this.artwork(frame, x, y, size, size, parent);
    } else {
      const symbols: Record<string, string> = { home: '⌂', tutorial: '▤', settings: '⚙', collection: '▧',
        locate: '✦', peek: '◉', addTime: '+', undo: '↶', back: '‹', next: '›', return: '↧', cancel: '×' };
      this.text(symbols[name] || '·', x, y, size, size, size * 0.65, C.text, false, parent);
    }
  }

  private iconButton(name: string, label: string, x: number, y: number, size: number,
    invoke: () => void, fill: string = C.surface): void {
    this.button('', x, y, size, size, invoke, fill, C.text, this.root, false, label);
    this.icon(name, x, y, size * 0.85);
  }

  private openTutorial(): void {
    this.clear(); this.screen = 'tutorial';
    this.tutorial = new TutorialScreen(this.root, this.height, this.top, this.bottom,
      () => this.renderHome(), { onStart: () => {
          if (this.save.game && this.save.game.attempt.status !== 'Won') this.renderGame();
          else this.enterLevel(unlockedThrough(this.save, this.levels.map(l => l.id)));
        },
        registerControl: (label, invoke, bounds) => this.bridge.control(label, invoke, bounds),
        clearControls: () => this.bridge.clear(), onText: value => this.bridge.text(value), reducedMotion: this.save.settings.reducedMotion });
  }

  private renderHome(): void {
    this.consumeTime(true);
    if (this.screen === 'game') this.persist();
    if (this.save.game?.attempt.status === 'PreviewingReference') {
      this.save.game = reduceGame(this.save.game, { type: 'endPeek' }).state; this.persist();
    }
    this.screen = 'home'; this.clear();
    const mainY = this.height - this.bottom - 206;
    if (this.homeFrame) {
      if (!this.homeArtFrame) {
        const source = this.homeFrame;
        const width = source.originalSize.width, height = source.originalSize.height;
        this.homeArtFrame = new SpriteFrame(); this.homeArtFrame.texture = source.texture;
        this.homeArtFrame.rect = new Rect(0, height * 0.18, width, height * 0.55);
        this.homeArtFrame.originalSize = new Size(width, height * 0.55);
      }
      const artTop = this.top + 144, artBottom = mainY - 49;
      const size = this.homeArtFrame.originalSize;
      const scale = Math.min(374 / size.width, Math.max(100, artBottom - artTop) / size.height);
      this.artwork(this.homeArtFrame, 195, (artTop + artBottom) / 2, size.width * scale, size.height * scale);
    }
    this.round(83, this.top + 21, 122, 36, 18, C.butter);
    this.text(`✦ ${this.save.game?.wallet.points || 0}`, 83, this.top + 21, 116, 30, 14, C.text, true);
    this.iconButton('settings', '设置', 344, this.top + 21, 44, () => this.openSettings());
    this.text('拾色之间', 195, this.top + 69, 304, 62, 38, C.text, true);
    this.text('100 道自然色彩 · 5 段拾色旅程', 195, this.top + 113, 304, 26, 12, C.textSecondary);
    const next = unlockedThrough(this.save, this.levels.map(l => l.id));
    const resume = this.save.game;
    const continuing = resume && resume.attempt.status !== 'Won' && (resume.attempt.elapsedMs > 0 || resume.attempt.movesSpent > 0);
    const levelNumber = continuing ? resume.level.number : next + 1;
    this.round(195, mainY + 5, 310, 62, 28, '#E7B5A2');
    this.button(continuing ? '继续游戏' : '开始游戏', 195, mainY, 310, 62,
      () => continuing ? this.renderGame() : this.enterLevel(next), C.primary, '#FFFFFF');
    this.text(`第 ${levelNumber} 关 · ${this.levels[levelNumber - 1]?.title || '自然的颜色'}`,
      195, mainY + 49, 320, 24, 12, C.textSecondary);
    const cards = [
      { icon: 'home', label: '关卡', fill: C.mint, action: () => { this.chapterPage = Math.floor(next / 20); this.renderLevels(); } },
      { icon: 'tutorial', label: '教程', fill: C.butter, action: () => this.openTutorial() },
      { icon: 'addTime', label: '道具', fill: C.lilac, action: () => this.renderShop() },
    ];
    cards.forEach((card, index) => {
      const x = 83 + index * 112, y = mainY + 129;
      this.round(x, y + 3, 100, 96, 22, '#EDE5DA');
      this.button('', x, y, 100, 96, card.action, card.fill, C.text, this.root, false, card.label);
      this.icon(card.icon, x, y - 13, 66);
      this.text(card.label, x, y + 28, 88, 24, 14, C.text, true);
    });
    this.text(`${this.save.completed.length} / 100`, 195, this.height - this.bottom - 9, 200, 18, 12, C.textSecondary);
  }

  private renderShop(): void {
    this.consumeTime(true); if (this.screen === 'game') this.persist(); this.screen = 'shop'; this.clear();
    this.iconButton('back', '返回首页', 40, this.top + 22, 44, () => this.renderHome());
    this.text('道具', 195, this.top + 20, 190, 36, 24, C.text, true);
    this.round(195, this.top + 80, 184, 44, 22, C.butter);
    this.text(`✦  ${this.save.game?.wallet.points || 0} 积分`, 195, this.top + 80, 180, 40, 17, C.text, true);
    const descriptions = {
      locate: ['点睛', '归位并固定 1 块颜色', '每关最多 2 次'],
      peek: ['窥色', '展示一条完整色路 3 秒', '每关最多 1 次'],
      addTime: ['续时', '为本关增加 30 秒', '每关最多 2 次'],
    };
    (['locate', 'peek', 'addTime'] as Prop[]).forEach((prop, index) => {
      const y = this.top + 175 + index * 133;
      this.round(195, y + 4, 342, 122, 24, '#EBE4D8');
      this.round(195, y, 342, 122, 24, C.surface);
      this.round(79, y - 10, 76, 76, 22, [C.mint, C.lilac, C.butter][index]);
      this.icon(prop, 79, y - 10, 68);
      this.text(descriptions[prop][0], 223, y - 36, 200, 28, 19, C.text, true);
      this.text(descriptions[prop][1], 231, y - 8, 216, 24, 13, C.textSecondary);
      this.text(`库存 ${this.save.game?.wallet.inventory[prop] || 0} · ${descriptions[prop][2]}`,
        217, y + 15, 216, 22, 11, C.textSecondary);
      this.button(`${PROP_PRICES[prop]} 积分购买`, 235, y + 44, 196, 32,
        () => this.buyProp(prop), C.primary, '#FFFFFF', this.root, false, `购买${descriptions[prop][0]}`);
    });
    this.text(this.notice || '通关获得积分，最佳成绩提升还能补发差额', 195,
      this.height - this.bottom - 24, 344, 40, 12, C.textSecondary);
  }

  private buyProp(prop: Prop, inGame = false): void {
    if (!this.save.game) return;
    const before = this.save.game;
    const result = reduceGame(before, { type: 'purchaseProp', prop });
    if (result.changed) {
      this.save.game = result.state;
      if (!this.persist()) { this.save.game = before; this.notice = '保存未成功，积分未扣除，请稍后再试'; }
      else this.notice = '已放入道具背包';
    } else this.notice = result.reason || '积分不足，通关后再来看看';
    if (inGame) { this.renderGame(); this.openProp(prop); }
    else this.renderShop();
  }

  private openSettings(): void {
    const { layer, top } = this.sheet('设置', '', 414);
    const labels = { sound: '轻音效', haptics: '轻触反馈', reducedMotion: '减少动效' };
    (['sound', 'haptics', 'reducedMotion'] as const).forEach((key, i) => {
      this.button(`${labels[key]}   ${this.save.settings[key] ? '已开启' : '已关闭'}`,
        195, top + 157 + i * 57, 280, 48, () => {
          this.save.settings[key] = !this.save.settings[key]; this.persist();
          this.renderHome(); this.openSettings();
        }, this.save.settings[key] ? C.mint : C.disabled, C.text, layer, true);
    });
    this.button('好了，继续拾色', 195, top + 348, 280, 48, () => this.renderHome(), C.primary, C.onPrimary, layer, true);
  }

  private renderLevels(): void {
    this.consumeTime(true);
    if (this.screen === 'game') this.persist();
    this.screen = 'levels';
    this.clear();
    this.button('‹', 40, this.top + 24, 48, 48, () => this.renderHome(), C.surface, C.text);
    const first = this.levels[this.chapterPage * 20];
    this.text(`第 ${this.chapterPage + 1} 章 · ${first.chapterName}`, 195, this.top + 26, 260, 40, 20, C.text, true);
    this.text('一关一关，拾起沿途的颜色', 195, this.top + 78, 320, 30, 15, C.textSecondary);
    const unlocked = unlockedThrough(this.save, this.levels.map(l => l.id));
    for (let i = 0; i < 20; i++) {
      const index = this.chapterPage * 20 + i;
      const level = this.levels[index];
      const allowed = index <= unlocked;
      const done = this.save.completed.includes(level.id);
      this.button(`${index + 1}${done ? ' ✓' : ''}`,
        72 + i % 4 * 82, this.top + 157 + Math.floor(i / 4) * 72, 62, 56,
        () => { if (allowed) this.enterLevel(index); else { this.notice = `完成第 ${index} 关后解锁`; this.renderLevels(); } },

    }

      if (this.chapterPage > 0) { this.chapterPage--; this.renderLevels(); }
    }, C.surface, C.text);
    this.button('下一章', 290, pagerY, 120, 48, () => {
      if (this.chapterPage < 4) { this.chapterPage++; this.renderLevels(); }
    }, C.surface, C.text);

  }

  private enterLevel(index: number): void {
    const allowed = unlockedThrough(this.save, this.levels.map(l => l.id));
    if (index < 0 || index >= this.levels.length || index > allowed) return;
    this.save.game = this.save.game
      ? startLevel(this.save.game, this.levels[index], this.nextId('attempt'))
      : createGame(this.levels[index], this.nextId('attempt'));
    this.selectedCell = null;
    this.selectedPool = null;
    this.notice = '';
    this.trayPage = 0;
    this.persist();
    this.renderGame();
  }

  private dispatch(action: Action): void {
    if (!this.save.game || this.actionBusy) return;
    this.consumeTime(true);
    const result = reduceGame(this.save.game, action);
    if (result.changed) {
      this.save.game = result.state;
      this.selectedCell = null; this.selectedPool = null;
      completeLevel(this.save, this.levels.map(l => l.id));
      if (['place', 'move', 'locate'].includes(action.type)) this.feedback.play(
        result.state.attempt.status === 'Won' ? 'complete' : 'place', this.save.settings.sound, this.save.settings.haptics);
      this.notice = '';
      this.persist();
    } else if (result.reason) {
      this.notice = this.friendlyReason(result.reason);
    }
    this.renderGame();
  }

  private friendlyReason(reason: string): string {
    if (/stock|inventory/i.test(reason)) return '这个道具用完啦';
    if (/limit|maximum/i.test(reason)) return '这次的使用次数已用完';
    if (/lock|anchor/i.test(reason)) return '小圆点标记的颜色已经固定';
    return reason;
  }

  private renderGame(): void {
    this.consumeTime(true);
    if (!this.save.game) { this.renderHome(); return; }
    this.screen = 'game';
    this.clear();
    const state = this.save.game;
    const { level, attempt } = state;
    const layout = boardLayout(this.height, this.top, this.bottom, level.grid.columns,
      level.grid.rows, attempt.pool.length);
    const { cellSize, pitch, boardStartX, boardStartY } = layout;
    this.iconButton('back', '返回首页', 40, this.top + 22, 44, () => {
      this.selectedCell = null; this.selectedPool = null; this.renderHome();
    });
    this.iconButton('settings', '关卡菜单', 350, this.top + 22, 44, () => this.openGameMenu());
    this.text(`${level.title}`, 195, this.top + 14, 244, 30, 19, C.text, true);
    this.text(`第 ${level.number} 关`, 152, this.top + 46, 106, 22, 12, C.textSecondary);
    this.round(253, this.top + 47, 92, 30, 15, C.butter);
    this.timerLabel = this.text('', 253, this.top + 47, 88, 30, 17, C.text, true).getComponent(Label);
    this.updateTimerLabel();
    const xy = (cellId: string) => {
      const cell = level.cells.find(c => c.id === cellId)!;
      return { x: boardStartX + cell.x * pitch, y: boardStartY + cell.y * pitch };
    };
    const groupWidth = Math.min(370, Math.max(264, layout.groupWidth + 30));
    this.round(195, layout.groupTop + layout.groupHeight / 2 + 5, groupWidth,
      layout.groupHeight + 36, 27, '#E8E3D7');
    this.round(195, layout.groupTop + layout.groupHeight / 2, groupWidth,
      layout.groupHeight + 36, 27, '#FFFEFB');
    this.round(195, layout.trayTop + layout.trayHeight / 2, Math.min(groupWidth - 12, layout.trayWidth + 18),
      layout.trayHeight + 16, 16, '#F0F1E9');
    const tracks = this.element('Tracks', 195, 0, 390, this.height);
    tracks.setPosition(0, this.height / 2);
    const g = tracks.addComponent(Graphics);
    g.strokeColor = new Color().fromHEX(C.track);
    g.lineWidth = 2;
    level.paths.forEach(path => {
      path.cellIds.forEach((id, i) => {
        const p = xy(id);
        if (!i) g.moveTo(p.x - 195, -p.y); else g.lineTo(p.x - 195, -p.y);
      });
      g.stroke();
    });
    const tileHex = (id: string | null) => level.tiles.find(t => t.id === id)?.hex || C.surface;
    const previewCells = attempt.preview
      ? level.paths.find(path => path.id === attempt.preview!.pathId)?.cellIds ?? [] : [];
    for (const cell of level.cells) {
      const p = xy(cell.id);
      const tile = attempt.board[cell.id];
      const locked = level.anchors.includes(cell.id) || attempt.auxiliaryLocks.includes(cell.id);
      const hex = previewCells.includes(cell.id) ? cell.targetHex : tileHex(tile);
      if (this.selectedCell === cell.id) this.round(p.x, p.y, cellSize + 8, cellSize + 8, 12, '#A6CEC3');
      const node = this.round(p.x, p.y, cellSize, cellSize, theme.radii.tile, hex,
        this.root, tile === null && !previewCells.includes(cell.id) ? '#DDD7CB' : undefined);
      node.on(Node.EventType.TOUCH_END, (event: EventTouch) => {
        event.propagationStopped = true;
        if (!this.modalOpen) this.tapCell(cell.id);
      });
      if (!locked) this.bridge.control(`上方${tile ? '色块' : '空槽'} ${cell.id}`, () => { if (!this.modalOpen) this.tapCell(cell.id); }, { x: p.x, y: p.y, width: cellSize, height: cellSize });
      if (locked) {
        this.round(p.x + cellSize / 2 + 1, p.y - cellSize / 2 - 1, 7, 7, 4, C.surface);
        const marker = this.round(p.x + cellSize / 2 + 1, p.y - cellSize / 2 - 1,
          4, 4, level.anchors.includes(cell.id) ? 2 : 0, C.text);

    // Instructions live in the interactive tutorial, not permanently on the play board.

    this.trayPage = Math.min(pages - 1, this.trayPage);

      const x = 195 - (layout.trayColumns - 1) * layout.trayPitch / 2

      const node = this.round(x, y, 44, 44, 8, tileHex(id), this.root, id ? undefined : C.border);
      node.on(Node.EventType.TOUCH_END, (event: EventTouch) => {
        event.propagationStopped = true;
        if (!this.modalOpen) this.tapPool(i);

      this.bridge.control(`候选槽 ${i + 1}${id ? ' ' + tileHex(id) : ' 空'}`, () => { if (!this.modalOpen) this.tapPool(i); }, { x, y, width: 44, height: 44 });
    }
    if (pages > 1) {
    this.iconButton('back', '上一页候选', 130, layout.pagerY, 40, () => {
      if (this.trayPage > 0) { this.trayPage--; this.renderGame(); }
    });
    this.text(`${this.trayPage + 1} / ${pages}`, 195, layout.pagerY, 70, 26, 12, C.textSecondary);
    this.iconButton('next', '下一页候选', 260, layout.pagerY, 40, () => {
      if (this.trayPage + 1 < pages) { this.trayPage++; this.renderGame(); }
    });
    }

      this.button('继续本关', 64, layout.actionsY, 96, 48, () => this.openFail(), C.mint, C.text);
      this.button('免费重开', 195, layout.actionsY, 144, 48,
        () => this.dispatch({ type: 'restart', attemptId: this.nextId('attempt') }), C.surface, C.text);
      this.button('选关', 326, layout.actionsY, 92, 48,
        () => { this.chapterPage = level.chapter - 1; this.renderLevels(); }, C.surface, C.text);
    } else {
    this.iconButton('cancel', '取消选择', 133, layout.actionsY, 44, () => {
      this.selectedCell = null; this.selectedPool = null; this.notice = ''; this.renderGame();
    }, this.selectedCell || this.selectedPool !== null ? C.surface : C.background);
    this.iconButton('return', '放回待选区', 195, layout.actionsY, 44, () => {
      if (this.selectedCell) this.dispatch({ type: 'returnToTray', fromCellId: this.selectedCell });
      else { this.notice = '先点选上方已填的颜色'; this.renderGame(); }
    }, this.selectedCell ? C.mint : C.background);
    this.iconButton('undo', '撤销', 257, layout.actionsY, 44,
      () => this.dispatch({ type: 'undo' }), C.surface);
    }
    (['locate', 'peek', 'addTime'] as Prop[]).forEach((prop, i) => {

        () => this.openProp(prop), attempt.status === 'Playing' ? [C.mint, C.lilac, C.butter][i] : C.disabled,
        C.text, this.root, false, names[prop]);

      this.round(x + 33, layout.propsY - 23, 22, 22, 11, C.surface);

  }

  private tapPool(index: number): void {
    this.consumeTime(true);
    const state = this.save.game!;
    if (state.attempt.status !== 'Playing') { this.renderGame(); return; }
    if (state.attempt.pool[index]) {
      this.selectedPool = this.selectedPool === index ? null : index;
      this.selectedCell = null; this.notice = ''; this.renderGame();
    } else if (this.selectedCell) {
      this.dispatch({ type: 'returnToTray', fromCellId: this.selectedCell, toPoolIndex: index });
    }
  }

  private tapCell(cellId: string): void {
    this.consumeTime(true);
    const state = this.save.game!;
    if (state.attempt.status !== 'Playing') { this.renderGame(); return; }
    if (state.level.anchors.includes(cellId) || state.attempt.auxiliaryLocks.includes(cellId)) {
      this.notice = '带小圆点的颜色已经固定'; this.renderGame(); return;
    }
    if (this.selectedCell === cellId) {
      this.selectedCell = null; this.renderGame(); return;
    }
    if (state.attempt.board[cellId] !== null) {
      if (this.selectedPool !== null || this.selectedCell !== null) {
        this.notice = '这里已有颜色，请先取回或选一个空格';
      } else { this.selectedCell = cellId; this.notice = ''; }
      this.renderGame(); return;
    }
    if (this.selectedPool !== null) {
      this.dispatch({ type: 'place', poolIndex: this.selectedPool, toCellId: cellId });
    } else if (this.selectedCell) {
      this.dispatch({ type: 'move', fromCellId: this.selectedCell, toCellId: cellId });
    } else { this.notice = '先点选下方的一块颜色'; this.renderGame(); }
  }

    tint.a = alpha;

  }

          if (entitlement) this.dispatch({ type: 'applyPendingRevive', requestId: entitlement.requestId });

        }, C.primary, '#FFFFFF', layer, true);
    } else this.text('本关复活次数已用完', 195, top + 196, 288, 56, 16, C.textSecondary, false, layer);

  }

  private openProp(prop: Prop): void {
    this.consumeTime(true);
    const state = this.save.game!;
    if (state.attempt.status !== 'Playing') { this.renderGame(); return; }

    const descriptions = { locate: '归位并固定 1 块颜色', peek: '选择一条色路，展示完整颜色 3 秒', addTime: '为本关倒计时增加 30 秒' };
    const { layer, top } = this.sheet(names[prop], descriptions[prop], 430);
    const limits = { locate: 2, peek: 1, addTime: 2 };
    const usesLeft = limits[prop] - state.attempt.propUses[prop];
    const stock = state.wallet.inventory[prop];
    this.text(`库存 ${stock} · 本关还可用 ${usesLeft} 次`, 195, top + 142, 290, 28, 14, C.textSecondary, false, layer);
    this.text(this.notice || `积分余额 ${state.wallet.points}`, 195, top + 173, 290, 26, 13, C.textSecondary, false, layer);
    if (usesLeft <= 0) this.text('本关使用次数已用完', 195, top + 225, 280, 48, 16, C.textSecondary, false, layer);
    else this.button(stock > 0 ? '使用道具' : `${PROP_PRICES[prop]} 积分购买 1 个`, 195, top + 225, 280, 50, () => {
      if (stock <= 0) { this.buyProp(prop, true); return; }
      this.notice = '';
      if (prop === 'peek') { this.openPathPicker(); return; }
      this.dispatch(prop === 'locate' ? { type: 'locate' } : { type: 'addTime' });
    }, C.primary, '#FFFFFF', layer, true);
    if (state.attempt.propRestockAdsGranted < 1 && stock === 0 && usesLeft > 0)
      this.button('▷ 看视频领取 1 个', 195, top + 293, 280, 48,
        () => this.requestVideo('restock', prop), C.butter, C.text, layer, true);
    else this.button(`积分购买 · ${PROP_PRICES[prop]} 分`, 195, top + 293, 280, 48,
      () => this.buyProp(prop, true), C.lilac, C.text, layer, true);
    this.button('继续游戏', 195, top + 365, 230, 44,
      () => { this.notice = ''; this.renderGame(); }, C.surface, C.text, layer, true);

  private openPathPicker(): void {
    this.renderGame();
    const paths = this.save.game!.level.paths;
    const height = Math.min(this.height - this.top - this.bottom - 24, 180 + Math.ceil(paths.length / 3) * 50);
    const { layer, top } = this.sheet('想看看哪条色路？', '参考 3 秒后自动收起', height);
    this.button('×', 327, top + 24, 40, 40, () => this.renderGame(), C.surface, C.text, layer, true);
    paths.forEach((path, i) => this.button(`色路 ${i + 1}`, 89 + i % 3 * 106, top + 153 + Math.floor(i / 3) * 50,
      96, 44, () => {
        this.dispatch({ type: 'peek', pathId: path.id, nowMs: Date.now() });
        const attemptId = this.save.game!.attempt.id;
        this.scheduleOnce(() => {
          if (this.save.game?.attempt.id !== attemptId) return;
          const result = reduceGame(this.save.game, { type: 'expirePeek', nowMs: Date.now() });
          if (result.changed) {
            this.save.game = result.state; this.persist();
            if (this.screen === 'game') this.renderGame();
          }
        }, 3);
      }, C.lilac, C.text, layer, true));
  }

  private requestVideo(rewardType: 'revive' | 'restock' | 'doubleScore', prop?: Prop): void {
    if (!this.video.available) {
      this.renderGame();
      const { layer, top } = this.sheet('广告暂不可用', '激励视频广告位尚未开通，开通后即可领取奖励', 280);
      this.button('返回游戏', 195, top + 197, 260, 48, () => this.renderGame(), C.mint, C.text, layer, true);
      return;
    }
    const requestId = this.nextId('reward');
    const before = this.save.game!;
    const result = reduceGame(before, { type: 'beginReward', requestId, rewardType, prop });
    if (!result.changed) { this.notice = result.reason || '本次奖励暂不可领取'; this.renderGame(); return; }
    this.save.game = result.state;
    if (!this.persist()) { this.save.game = before; this.renderGame(); return; }
    this.renderGame();
    this.sheet('正在准备视频', '完整观看后发放奖励；提前关闭不会扣除道具或积分', 240);
    this.video.show(outcome => {
      if (!this.save.game) return;
      const settled = reduceGame(this.save.game, { type: 'settleReward', requestId, outcome });
      this.save.game = settled.state;
      this.notice = outcome === 'completed' ? '' : outcome === 'failed'
        ? '视频暂不可用，盘面和库存已保留' : '未完整观看，本次没有领取奖励';
      if (outcome === 'completed') this.persistConfirmedReward();
      else this.persist();
      if (this.screen === 'game') this.renderGame();
      else if (this.screen === 'shop') this.renderShop();
      else this.renderHome();
    });
  }
}

/** Pure game rules. No Cocos, platform SDK, clock, randomness or storage dependencies. */
export const RULES_VERSION = 'click-fill-timed-flex-v3' as const;
export type Prop = 'locate' | 'peek' | 'addTime';
export type Inventory = Record<Prop, number>;
export type Board = Record<string, string | null>;
export type AttemptStatus = 'Playing' | 'Won' | 'Failed' | 'AdPending' | 'PreviewingReference';
export interface LevelDefinition {
  gameplayMode: 'click-fill';
  inputMode: 'tap';
  id: string;
  number: number;
  title: string;
  chapter: number;
  chapterName: string;
  contentVersion: string;
  /** Different puzzle designs only compare performance within the same scoring epoch. */
  scoreEpoch?: string;
  grid: { columns: number; rows: number };
  cells: { id: string; x: number; y: number; targetHex: string }[];
  paths: { id: string; cellIds: string[] }[];
  anchors: string[];
  tiles: { id: string; hex: string }[];
  initialBoard: Board;
  initialPool: string[];
  /** Authored initial slot indexes; movable tiles may occupy any empty tray slot at runtime. */
  trayHomeByTile: Record<string, number>;
  tutorialText: string;
  challenge: { mode: 'time'; timeLimitSeconds: number; revive: { maxPerAttempt: number; addSeconds: number } };
}
export interface UndoEntry {
  kind: 'place' | 'move' | 'returnToTray' | 'locate';
  board: Board;
  pool: (string | null)[];
  auxiliaryLocks: string[];
}
export interface Attempt {
  rulesVersion: typeof RULES_VERSION;
  id: string;
  levelId: string;
  levelContentVersion: string;
  revision: number;
  status: AttemptStatus;
  board: Board;
  pool: (string | null)[];
  auxiliaryLocks: string[];
  initialTimeMs: number;
  remainingTimeMs: number;

export const DEFAULT_SCORE_EPOCH = 'base-v1';
export const PROP_LIMITS: Readonly<Inventory> = Object.freeze({ locate: 2, peek: 1, addTime: 2 });
export const PROP_PRICES: Readonly<Inventory> = Object.freeze({ locate: 120, peek: 80, addTime: 100 });
export const INITIAL_INVENTORY: Readonly<Inventory> = Object.freeze({ locate: 1, peek: 1, addTime: 1 });
export const STARTER_INVENTORY: Readonly<Inventory> = Object.freeze({ locate: 3, peek: 2, addTime: 1 });
const own = (object: object, key: string): boolean => Object.prototype.hasOwnProperty.call(object, key);
const emptyInventory = (): Inventory => ({ locate: 0, peek: 0, addTime: 0 });
const hex = (value: unknown): string => {
  if (typeof value !== 'string' || !/^#[0-9a-f]{6}$/i.test(value)) throw new Error('Invalid HEX colour');
  return value.toUpperCase();
};
const object = (value: unknown, name: string): Record<string, unknown> => {
  if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error(`Invalid ${name}`);
  return value as Record<string, unknown>;
};
const array = (value: unknown, name: string): unknown[] => {
  if (!Array.isArray(value)) throw new Error(`Invalid ${name}`);
  return value;
};
const string = (value: unknown, name: string): string => {
  if (typeof value !== 'string' || value.length === 0) throw new Error(`Invalid ${name}`);
  return value;
};
const integer = (value: unknown, name: string, minimum = 0): number => {
  if (typeof value !== 'number' || !Number.isSafeInteger(value) || value < minimum) throw new Error(`Invalid ${name}`);
  return value;
};

/** Strict boundary adapter for the accepted candidate pack; it never repairs content. */
export function adaptLevel(input: unknown): LevelDefinition {
  const raw = object(input, 'level');
  if (raw.gameplayMode !== 'click-fill' || raw.inputMode !== 'tap') throw new Error('Only click-fill levels are supported');
  const grid = object(raw.grid, 'grid');
  const challenge = object(raw.challenge, 'challenge');
  const revive = object(challenge.revive, 'revive');
  const solution = object(raw.solution, 'solution');
  if (array(solution.acceptedAlternativeLayouts, 'alternative layouts').length !== 0) {
    throw new Error('Alternative answer layouts require an explicit adapter upgrade');
  }
  if (!['none', 'moves', 'time'].includes(challenge.mode as string)) throw new Error('Invalid challenge mode');
  const targetByCell = object(solution.targetByCell, 'targetByCell');
  const cells = array(raw.cells, 'cells').map(value => {
    const cell = object(value, 'cell');
    const id = string(cell.id, 'cell id');
    const targetHex = hex(cell.targetHex);
    if (hex(targetByCell[id]) !== targetHex) throw new Error('Cell and solution disagree');
    return { id, x: integer(cell.x, 'cell x'), y: integer(cell.y, 'cell y'), targetHex };
  });
  const tiles = array(raw.tiles, 'tiles').map(value => {
    const tile = object(value, 'tile');
    return { id: string(tile.id, 'tile id'), hex: hex(tile.hex) };
  });
  const paths = array(raw.paths, 'paths').map(value => {
    const path = object(value, 'path');
    return { id: string(path.id, 'path id'), cellIds: array(path.cellIds, 'path cells').map(v => string(v, 'path cell')) };
  });
  const initialBoardRaw = object(raw.initialBoard, 'initialBoard');
  const initialBoard: Board = Object.create(null) as Board;
  for (const cell of cells) {
    const value = initialBoardRaw[cell.id];
    initialBoard[cell.id] = value === null ? null : string(value, 'board tile');
  }
  const level: LevelDefinition = {
    gameplayMode: 'click-fill', inputMode: 'tap',
    id: string(raw.id, 'level id'), number: integer(raw.number, 'level number', 1),
    title: string(raw.title, 'level title'), chapter: integer(raw.chapter, 'chapter', 1),
    chapterName: string(raw.chapterName, 'chapterName'), contentVersion: string(raw.contentVersion, 'contentVersion'),
    scoreEpoch: raw.scoreEpoch === undefined ? DEFAULT_SCORE_EPOCH : string(raw.scoreEpoch, 'score epoch'),
    grid: { columns: integer(grid.columns, 'columns', 1), rows: integer(grid.rows, 'rows', 1) },
    cells, paths, anchors: array(raw.anchors, 'anchors').map(v => string(v, 'anchor')),
    tiles, initialBoard, initialPool: array(raw.initialPool, 'pool').map(v => string(v, 'pool tile')),
    trayHomeByTile: Object.fromEntries(Object.entries(object(raw.trayHomeByTile, 'trayHomeByTile')).map(([id, index]) => [id, integer(index, 'home slot')])),
    tutorialText: typeof raw.tutorialText === 'string' ? raw.tutorialText : '',
    challenge: {
      mode: 'time',
      timeLimitSeconds: challenge.mode === 'time' ? integer(challenge.timeLimitSeconds, 'time limit', 1)
        : defaultTimeLimitSeconds(integer(raw.number, 'number', 1), integer(raw.chapter, 'chapter', 1), array(raw.initialPool, 'pool').length),
      revive: { maxPerAttempt: 2, addSeconds: 30 },
    },
  };
  if (!cells.length || new Set(cells.map(c => c.id)).size !== cells.length) throw new Error('Duplicate or empty cells');
  if (new Set(cells.map(c => `${c.x}:${c.y}`)).size !== cells.length) throw new Error('Duplicate coordinates');
  if (cells.some(c => c.x >= level.grid.columns || c.y >= level.grid.rows)) throw new Error('Cell outside grid');
  if (new Set(tiles.map(t => t.id)).size !== tiles.length) throw new Error('Duplicate tile IDs');
  if (new Set(paths.map(p => p.id)).size !== paths.length) throw new Error('Duplicate path IDs');
  if (new Set(level.anchors).size !== level.anchors.length) throw new Error('Duplicate anchors');
  if (paths.some(p => !p.cellIds.length || p.cellIds.some(id => !own(initialBoard, id)))) throw new Error('Invalid path reference');
  if (Object.keys(initialBoardRaw).length !== cells.length || Object.keys(targetByCell).length !== cells.length) throw new Error('Unexpected board cell');
  if (level.anchors.some(id => !own(initialBoard, id))) throw new Error('Invalid anchor reference');
  const allTiles = [...Object.values(initialBoard), ...level.initialPool].filter((v): v is string => v !== null);
  const tileIds = new Set(tiles.map(t => t.id));
  if (allTiles.length !== tiles.length || new Set(allTiles).size !== tiles.length || allTiles.some(id => !tileIds.has(id))) {
    throw new Error('Initial tile conservation failed');
  }
  if (cells.length !== tiles.length) throw new Error('One tile per target cell required');
  const targets = cells.map(c => c.targetHex).sort();
  const colours = tiles.map(t => t.hex).sort();
  if (targets.some((colour, i) => colour !== colours[i])) throw new Error('Target colour multiset mismatch');
  if (level.anchors.some(id => colourAt(level, initialBoard, id) !== cells.find(c => c.id === id)!.targetHex)) {
    throw new Error('Initial anchor is not correct');
  }
  if (challenge.mode === 'time' && (revive.maxPerAttempt !== 2 || revive.addSeconds !== 30)) throw new Error('Invalid timed revival policy');
  if (cells.some(cell => !level.anchors.includes(cell.id) && initialBoard[cell.id] !== null)) throw new Error('Only anchor tiles may be prefilled');
  const movableCount = cells.length - level.anchors.length;
  if (movableCount < 1 || level.initialPool.length !== movableCount) throw new Error('All movable tiles must start in the tray');
  if (Object.keys(level.trayHomeByTile).length !== movableCount || level.initialPool.some((id, index) => level.trayHomeByTile[id] !== index)) throw new Error('Invalid permanent tray homes');
  if (challenge.minimumMovesMetric !== 'tray-to-empty-board-placements' || challenge.minimumMoves !== movableCount) throw new Error('Invalid minimum placement metric');
  if (level.challenge.timeLimitSeconds < 45 || level.challenge.timeLimitSeconds > 240) throw new Error('Time limit outside supported range');
  return level;
}

function colourAt(level: LevelDefinition, board: Board, cellId: string): string | null {
  const tileId = board[cellId];
  return tileId == null ? null : level.tiles.find(tile => tile.id === tileId)?.hex ?? null;
}
export function isSolved(level: LevelDefinition, board: Board, pool: (string | null)[]): boolean {
  return pool.length === level.initialPool.length && pool.every(id => id === null)
    && level.cells.every(cell => colourAt(level, board, cell.id) === cell.targetHex);
}
export function isLocked(state: GameState, cellId: string): boolean {
  return state.level.anchors.includes(cellId) || state.attempt.auxiliaryLocks.includes(cellId);
}

function makeAttempt(level: LevelDefinition, attemptId: string): Attempt {
  string(attemptId, 'attempt ID');
  return {
    rulesVersion: RULES_VERSION, id: attemptId, levelId: level.id, levelContentVersion: level.contentVersion, revision: 0,
    status: isSolved(level, level.initialBoard, level.initialPool) ? 'Won' : 'Playing',
    board: { ...level.initialBoard }, pool: [...level.initialPool], auxiliaryLocks: [],
    initialTimeMs: level.challenge.timeLimitSeconds * 1000, remainingTimeMs: level.challenge.timeLimitSeconds * 1000, elapsedMs: 0,
    movesSpent: 0, totalTimeGrantedMs: 0, score: null, pointsEarned: 0, undoStack: [], propUses: emptyInventory(),
    revivesUsed: 0, propRestockAdsGranted: 0, reviveAdsGranted: 0,
    pendingRewardRequestId: null, previousStatusBeforeAd: null, preview: null,
  };
}
function validateInventory(inventory: Inventory): Inventory {
  return { locate: integer(inventory.locate, 'locate stock'), peek: integer(inventory.peek, 'peek stock'), addTime: integer(inventory.addTime, 'addTime stock') };
}
export function createGame(level: LevelDefinition, attemptId: string, inventory: Inventory = { ...INITIAL_INVENTORY }): GameState {
  return { level, attempt: makeAttempt(level, attemptId), wallet: { points: 0, levelScores: {}, inventory: validateInventory(inventory), rewardLedger: {}, pendingRevives: [] } };
}

function restoreLayout(level: LevelDefinition, input: unknown): Pick<UndoEntry, 'board' | 'pool' | 'auxiliaryLocks'> {
  const raw = object(input, 'layout');
  const rawBoard = object(raw.board, 'saved board');
  if (Object.keys(rawBoard).length !== level.cells.length) throw new Error('Saved board size mismatch');
  const board: Board = {};
  for (const cell of level.cells) board[cell.id] = rawBoard[cell.id] === null ? null : string(rawBoard[cell.id], 'saved tile');
  const pool = array(raw.pool, 'saved tray').map(value => value === null ? null : string(value, 'saved tray tile'));
  const auxiliaryLocks = array(raw.auxiliaryLocks, 'saved locks').map(value => string(value, 'saved lock'));
  if (pool.length !== level.initialPool.length) throw new Error('Saved tray size mismatch');
  const ids = [...Object.values(board), ...pool].filter((id): id is string => id !== null);
  const tileIds = new Set(level.tiles.map(tile => tile.id));
  if (ids.length !== tileIds.size || new Set(ids).size !== tileIds.size || ids.some(id => !tileIds.has(id))) throw new Error('Saved tile conservation failed');
  if (level.anchors.some(id => board[id] !== level.initialBoard[id])) throw new Error('Saved anchor moved');
  if (new Set(auxiliaryLocks).size !== auxiliaryLocks.length || auxiliaryLocks.some(id => !own(board, id) || level.anchors.includes(id) || colourAt(level, board, id) !== level.cells.find(cell => cell.id === id)!.targetHex)) throw new Error('Saved lock invalid');
  return { board, pool, auxiliaryLocks };
}

/** Restores only this rules/content version. Old attempts must be discarded, never reinterpreted. */
export function restoreAttempt(level: LevelDefinition, input: unknown): Attempt {
  const raw = object(input, 'attempt');
  if (raw.rulesVersion !== RULES_VERSION || raw.levelId !== level.id || raw.levelContentVersion !== level.contentVersion) throw new Error('Incompatible attempt rules or level version');
  const layout = restoreLayout(level, raw);
  const propUses = validateInventory(object(raw.propUses, 'saved prop uses') as Inventory);
  for (const prop of Object.keys(PROP_LIMITS) as Prop[]) if (propUses[prop] > PROP_LIMITS[prop]) throw new Error('Saved prop limit exceeded');
  const initialTimeMs = integer(raw.initialTimeMs, 'saved initial time', 1);
  const remainingTimeMs = integer(raw.remainingTimeMs, 'saved remaining time');
  const elapsedMs = integer(raw.elapsedMs, 'saved elapsed time');
  const totalTimeGrantedMs = integer(raw.totalTimeGrantedMs, 'saved granted time');
  const movesSpent = integer(raw.movesSpent, 'saved spent moves');
  if (initialTimeMs !== level.challenge.timeLimitSeconds * 1000) throw new Error('Saved initial time mismatch');
  if (remainingTimeMs !== initialTimeMs + totalTimeGrantedMs - elapsedMs) throw new Error('Saved time accounting mismatch');
  const score = raw.score === null ? null : integer(raw.score, 'saved score', 100);
  if (score !== null && score > 200) throw new Error('Saved score too high');
  const pointsEarned = integer(raw.pointsEarned, 'saved points earned');
  if (pointsEarned > 200 || (score === null && pointsEarned !== 0)) throw new Error('Saved score award invalid');
  const revivesUsed = integer(raw.revivesUsed, 'saved revival count');
  const propRestockAdsGranted = integer(raw.propRestockAdsGranted, 'saved restock count');
  const reviveAdsGranted = integer(raw.reviveAdsGranted, 'saved rewarded revival count');
  if (totalTimeGrantedMs !== (propUses.addTime + revivesUsed) * 30000) throw new Error('Saved granted time mismatch');
  if (revivesUsed > level.challenge.revive.maxPerAttempt || propRestockAdsGranted > 1 || reviveAdsGranted > revivesUsed) throw new Error('Saved reward limit exceeded');
  const status = raw.status as AttemptStatus;
  if (!['Playing', 'Won', 'Failed', 'AdPending', 'PreviewingReference'].includes(status)) throw new Error('Saved status invalid');
  const solved = isSolved(level, layout.board, layout.pool);
  const previousStatusBeforeAd = raw.previousStatusBeforeAd === null ? null : raw.previousStatusBeforeAd as 'Playing' | 'Failed' | 'Won';
  const pendingRewardRequestId = raw.pendingRewardRequestId === null ? null : string(raw.pendingRewardRequestId, 'pending request');
  const baseStatus = status === 'AdPending' ? previousStatusBeforeAd : status === 'PreviewingReference' ? 'Playing' : status;
  if (!['Playing', 'Won', 'Failed'].includes(baseStatus ?? '')) throw new Error('Saved base status invalid');
  if (solved !== (baseStatus === 'Won') || (baseStatus === 'Failed' && remainingTimeMs !== 0) || (baseStatus === 'Playing' && remainingTimeMs === 0)) throw new Error('Saved status disagrees with board or budget');
  if ((baseStatus === 'Won') !== (score !== null)) throw new Error('Saved score status mismatch');
  if ((status === 'AdPending') !== (pendingRewardRequestId !== null && previousStatusBeforeAd !== null)) throw new Error('Saved ad state invalid');
  let preview: Attempt['preview'] = null;
  if (status === 'PreviewingReference') {
    const item = object(raw.preview, 'saved preview');
    const pathId = string(item.pathId, 'saved preview path');
    if (!level.paths.some(path => path.id === pathId) || typeof item.expiresAtMs !== 'number' || !Number.isFinite(item.expiresAtMs)) throw new Error('Saved preview invalid');
    preview = { pathId, expiresAtMs: item.expiresAtMs };
  } else if (raw.preview !== null) throw new Error('Unexpected saved preview');
  const undoStack = array(raw.undoStack, 'saved history').map(value => {
    const entry = object(value, 'saved history entry');
    if (!['place', 'move', 'returnToTray', 'locate'].includes(entry.kind as string)) throw new Error('Incompatible history action');
    return { kind: entry.kind as UndoEntry['kind'], ...restoreLayout(level, entry) };
  });
  return { ...layout, rulesVersion: RULES_VERSION, id: string(raw.id, 'saved attempt ID'), levelId: level.id,
    levelContentVersion: level.contentVersion, revision: integer(raw.revision, 'saved revision'), status,
    initialTimeMs, remainingTimeMs, elapsedMs, movesSpent, totalTimeGrantedMs, score, pointsEarned, undoStack,
    propUses, revivesUsed, propRestockAdsGranted, reviveAdsGranted, pendingRewardRequestId, previousStatusBeforeAd, preview };
}

/** The application may preserve wallet/progress separately when an incompatible attempt is discarded. */
export function restoreGame(level: LevelDefinition, input: unknown): GameState {
  const raw = object(input, 'saved game');
  const attempt = restoreAttempt(level, raw.attempt);
  const wallet = restoreWallet(raw.wallet);
  if (attempt.pendingRewardRequestId) {
    const request = wallet.rewardLedger[attempt.pendingRewardRequestId];
    if (!request || request.status !== 'pending' || request.attemptId !== attempt.id || request.levelId !== level.id || request.levelContentVersion !== level.contentVersion) throw new Error('Saved pending reward mismatch');
  }
  return { level, attempt, wallet };
}

/** Validates earned rights independently of the attempt. Legacy move entitlements become seconds. */
export function restoreWallet(input: unknown, legacy = false): Wallet {
  const rawWallet = object(input, 'saved wallet');
  const rawInventory = object(rawWallet.inventory, 'saved inventory');
  const inventory = validateInventory({ ...rawInventory,
    addTime: rawInventory.addTime ?? (legacy ? rawInventory.addMoves : undefined),
  } as Inventory);
  const points = legacy && rawWallet.points === undefined ? 0 : integer(rawWallet.points, 'saved points');
  const levelScores: Record<string, LevelScore> = {};
  for (const [id, value] of Object.entries(legacy && rawWallet.levelScores === undefined ? {} : object(rawWallet.levelScores, 'level scores'))) {
    const item = object(value, 'level score');
    const bestScore = integer(item.bestScore, 'best score', 100);
    if (!/^L\d{3}$/.test(id) || bestScore > 200 || typeof item.doubleScoreClaimed !== 'boolean') throw new Error('Saved level score invalid');
    const scoreEpoch = item.scoreEpoch === undefined ? {} : { scoreEpoch: string(item.scoreEpoch, 'saved score epoch') };
    levelScores[id] = { ...scoreEpoch, bestScore, doubleScoreClaimed: item.doubleScoreClaimed };
  }
  const rewardLedger: Record<string, RewardRequest> = {};
  for (const [id, value] of Object.entries(object(rawWallet.rewardLedger, 'reward ledger'))) {
    const item = object(value, 'reward request');
    if (!validId(id) || item.id !== id || !['pending', 'granted', 'cancelled', 'failed'].includes(item.status as string) || !['revive', 'restock', 'doubleScore'].includes(item.rewardType as string)) throw new Error('Saved reward request invalid');
    const rewardType = item.rewardType as RewardRequest['rewardType'];
    const prop = rewardType === 'restock' ? (legacy && item.prop === 'addMoves' ? 'addTime' : item.prop) as Prop : null;
    const oldAmount = integer(item.amount, 'saved reward amount', 1);
    const amount = legacy && rewardType === 'revive' ? 30 : oldAmount;
    if (rewardType === 'restock' && (!own(PROP_LIMITS, prop!) || amount !== 1)) throw new Error('Saved restock reward invalid');
    if (rewardType === 'revive' && amount !== 30) throw new Error('Saved revival amount invalid');
    if (rewardType === 'doubleScore' && amount > 200) throw new Error('Saved bonus amount invalid');
    rewardLedger[id] = { id, attemptId: string(item.attemptId, 'reward attempt'), levelId: string(item.levelId, 'reward level'),
      levelContentVersion: string(item.levelContentVersion, 'reward content version'), rewardType, prop, amount,
      ...(item.scoreEpoch === undefined ? {} : { scoreEpoch: string(item.scoreEpoch, 'reward score epoch') }), status: item.status as RewardRequest['status'] };
  }
  if (Object.values(rewardLedger).filter(item => item.status === 'pending').length > 1) throw new Error('Multiple pending rewards');
  const pendingRevives = array(rawWallet.pendingRevives, 'pending revivals').map(value => {
    const item = object(value, 'pending revival');
    const requestId = string(item.requestId, 'pending revival request');
    const request = rewardLedger[requestId];
    const originAttemptId = string(item.originAttemptId, 'pending revival attempt');
    const oldAmount = integer(item.amount, 'pending revival amount', 1);
    const amount = legacy ? 30 : oldAmount;
    if (!request || request.status !== 'granted' || request.rewardType !== 'revive' || request.attemptId !== originAttemptId || request.amount !== amount) throw new Error('Saved pending revival mismatch');
    return { requestId, originAttemptId, amount };
  });
  if (new Set(pendingRevives.map(item => item.requestId)).size !== pendingRevives.length) throw new Error('Duplicate pending revival');
  return { points, levelScores, inventory, rewardLedger, pendingRevives };
}
/** Application owns sequential level unlocks and one-time starter/chapter grants. */
export function startLevel(previous: GameState, level: LevelDefinition, attemptId: string): GameState {
  if (attemptId === previous.attempt.id || Object.values(previous.wallet.rewardLedger).some(request => request.attemptId === attemptId)) {
    throw new Error('A new attempt ID is required');
  }
  return { level, attempt: makeAttempt(level, attemptId), wallet: cloneWallet(previous.wallet) };
}
/** Resume presentation-independent state. Unverified ad requests remain in the ledger. */
export function recoverInterruptedGame(state: GameState): GameState {
  if (state.attempt.status !== 'PreviewingReference' && state.attempt.status !== 'AdPending') return state;
  const next = clone(state);
  if (next.attempt.status === 'PreviewingReference') {
    next.attempt.preview = null;
    next.attempt.status = 'Playing';
  } else {
    next.attempt.status = next.attempt.previousStatusBeforeAd ?? 'Failed';
    next.attempt.previousStatusBeforeAd = null;
    next.attempt.pendingRewardRequestId = null;
  }
  next.attempt.revision++;
  return next;
}
function cloneWallet(wallet: Wallet): Wallet {
  const rewardLedger: Record<string, RewardRequest> = {};
  for (const id of Object.keys(wallet.rewardLedger)) rewardLedger[id] = { ...wallet.rewardLedger[id] };
  return { points: wallet.points, levelScores: Object.fromEntries(Object.entries(wallet.levelScores).map(([id, score]) => [id, { ...score }])), inventory: { ...wallet.inventory }, rewardLedger, pendingRevives: wallet.pendingRevives.map(item => ({ ...item })) };
}
function clone(state: GameState): GameState {
  const attempt = state.attempt;
  return {
    level: state.level,
    attempt: { ...attempt, board: { ...attempt.board }, pool: [...attempt.pool], auxiliaryLocks: [...attempt.auxiliaryLocks],
      undoStack: [...attempt.undoStack], propUses: { ...attempt.propUses }, preview: attempt.preview ? { ...attempt.preview } : null },
    wallet: cloneWallet(state.wallet),
  };
}
function snapshot(state: GameState, kind: UndoEntry['kind']): void {
  state.attempt.undoStack.push({ kind, board: { ...state.attempt.board }, pool: [...state.attempt.pool], auxiliaryLocks: [...state.attempt.auxiliaryLocks] });
}
function rejected(state: GameState, reason: string): Transition {
  return { state, changed: false, reason, events: [] };
}
function committed(state: GameState, events: DomainEvent[] = []): Transition {
  state.attempt.revision++;
  return { state, changed: true, events };
}
export function defaultTimeLimitSeconds(number: number, chapter: number, movableCount: number): number {
  return number === 1 ? 45 : Math.min(240, Math.max(45, 45 + movableCount * (chapter <= 2 ? 6 : chapter <= 4 ? 5 : 4)));
}
/** 100 base + 0..50 placement efficiency + 0..50 remaining original time, with aid penalties. */
export function calculateScore(state: GameState): number {
  const attempt = state.attempt;
  const minimum = state.level.initialPool.length;
  const efficiency = Math.round(50 * Math.min(1, minimum / Math.max(minimum, attempt.movesSpent)));
  const time = Math.round(50 * Math.min(1, Math.max(0, attempt.remainingTimeMs / attempt.initialTimeMs)));
  const aids = Object.values(attempt.propUses).reduce((sum, uses) => sum + uses, 0) + attempt.revivesUsed;
  return Math.max(100, 100 + efficiency + time - aids * 10);
}
function effectiveScoreEpoch(value: { scoreEpoch?: string }): string {
  return value.scoreEpoch ?? DEFAULT_SCORE_EPOCH;
}
export function canClaimDoubleScore(state: GameState): boolean {
  const record = state.wallet.levelScores[state.level.id];
  return state.attempt.status === 'Won' && state.attempt.pointsEarned > 0 && !!record
    && effectiveScoreEpoch(record) === effectiveScoreEpoch(state.level) && !record.doubleScoreClaimed;
}
function resolveBoard(state: GameState, spendMove: boolean): DomainEvent[] {
  const attempt = state.attempt;
  if (spendMove) attempt.movesSpent++;
  if (isSolved(state.level, attempt.board, attempt.pool)) {
    attempt.status = 'Won';
    attempt.score = calculateScore(state);
    const saved = state.wallet.levelScores[state.level.id];
    const previous = saved && effectiveScoreEpoch(saved) === effectiveScoreEpoch(state.level) ? saved : undefined;
    const bestScore = Math.max(previous?.bestScore ?? 0, attempt.score);
    attempt.pointsEarned = bestScore - (previous?.bestScore ?? 0);
    state.wallet.points += attempt.pointsEarned;
    state.wallet.levelScores[state.level.id] = { scoreEpoch: effectiveScoreEpoch(state.level), bestScore, doubleScoreClaimed: previous?.doubleScoreClaimed ?? false };
    return [{ type: 'won', levelId: state.level.id, attemptId: attempt.id, score: attempt.score, pointsEarned: attempt.pointsEarned }];
  }
  attempt.status = 'Playing';
  return [];
}
function propReason(state: GameState, prop: Prop): string | undefined {
  if (state.attempt.status !== 'Playing') return '当前状态不能使用道具';
  if (state.attempt.propUses[prop] >= PROP_LIMITS[prop]) return '本次尝试已达道具使用上限';
  if (state.wallet.inventory[prop] <= 0) return '道具库存不足';
  return undefined;
}
function spendProp(state: GameState, prop: Prop): void {
  state.wallet.inventory[prop]--;
  state.attempt.propUses[prop]++;
}
function validId(value: string): boolean {
  return typeof value === 'string' && value.length > 0 && value !== '__proto__' && value !== 'constructor' && value !== 'prototype';
}

/** Returns a new atomic snapshot. Persist the entire snapshot together, before animations. */
export function reduceGame(state: GameState, action: Action): Transition {
  const attempt = state.attempt;
  if (action.type === 'restart') {
    if (!validId(action.attemptId) || action.attemptId === attempt.id || Object.values(state.wallet.rewardLedger).some(r => r.attemptId === action.attemptId)) {
      return rejected(state, '重开必须使用新的尝试 ID');
    }
    const next = startLevel(state, state.level, action.attemptId);
    return committed(next);
  }
  if (action.type === 'settleReward') return settleReward(state, action.requestId, action.outcome);
  if (action.type === 'beginReward') return beginReward(state, action);
  if (action.type === 'applyPendingRevive') {
    if (attempt.status !== 'Failed' || attempt.revivesUsed >= state.level.challenge.revive.maxPerAttempt) return rejected(state, '当前不能使用待用复活权益');
    const item = state.wallet.pendingRevives.find(r => r.requestId === action.requestId);
    if (!item) return rejected(state, '待用复活权益不存在');
    const next = clone(state);
    next.wallet.pendingRevives = next.wallet.pendingRevives.filter(r => r.requestId !== action.requestId);
    applyRevive(next, item.amount);
    return committed(next);
  }
  if (action.type === 'endPeek' || action.type === 'expirePeek' || action.type === 'suspend') {
    if (attempt.status !== 'PreviewingReference') return rejected(state, '没有正在显示的参考');
    if (action.type === 'expirePeek' && (!Number.isFinite(action.nowMs) || action.nowMs < attempt.preview!.expiresAtMs)) return rejected(state, '参考尚未到期');
    const next = clone(state);
    next.attempt.status = 'Playing';
    next.attempt.preview = null;
    return committed(next);
  }
  if (action.type === 'purchaseProp') {
    if (attempt.status === 'AdPending' || attempt.status === 'PreviewingReference') return rejected(state, '请先结束当前操作');
    if (!own(PROP_PRICES, action.prop)) return rejected(state, '无效道具');
    const price = PROP_PRICES[action.prop];
    if (state.wallet.points < price) return rejected(state, '积分不足');
    const next = clone(state);
    next.wallet.points -= price;
    next.wallet.inventory[action.prop]++;
    return committed(next);
  }
  if (attempt.status !== 'Playing') return rejected(state, '当前状态不能操作棋盘');
  if (action.type === 'elapse') {
    if (!Number.isSafeInteger(action.deltaMs) || action.deltaMs <= 0) return rejected(state, '无效计时增量');
    const next = clone(state);
    const elapsed = Math.min(action.deltaMs, next.attempt.remainingTimeMs);
    next.attempt.remainingTimeMs -= elapsed;
    next.attempt.elapsedMs += elapsed;
    if (next.attempt.remainingTimeMs === 0) {
      next.attempt.status = 'Failed';
      return committed(next, [{ type: 'failed', levelId: next.level.id, attemptId: next.attempt.id }]);
    }
    return committed(next);
  }
  if (action.type === 'move') {
    const from = action.fromCellId, to = action.toCellId;
    if (from === to || !own(attempt.board, from) || !own(attempt.board, to) || isLocked(state, from) || isLocked(state, to) || !attempt.board[from] || attempt.board[to] !== null) {
      return rejected(state, '请选择色块和另一个空格');
    }
    const next = clone(state);
    snapshot(next, 'move');
    next.attempt.board[to] = next.attempt.board[from];
    next.attempt.board[from] = null;
    return committed(next, resolveBoard(next, true));
  }
  if (action.type === 'returnToTray') {
    const from = action.fromCellId;
    if (!own(attempt.board, from) || isLocked(state, from) || !attempt.board[from]) return rejected(state, '请选择可取回的色块');
    const home = state.level.trayHomeByTile[attempt.board[from]!];
    const index = action.toPoolIndex !== undefined ? action.toPoolIndex
      : attempt.pool[home] === null ? home : attempt.pool.findIndex(tile => tile === null);
    if (!Number.isInteger(index) || index < 0 || index >= attempt.pool.length || attempt.pool[index] !== null) return rejected(state, '请选择待选区的空槽');
    const next = clone(state);
    snapshot(next, 'returnToTray');
    next.attempt.pool[index] = next.attempt.board[from];
    next.attempt.board[from] = null;
    return committed(next, resolveBoard(next, false));
  }
  if (action.type === 'place') {
    const index = action.poolIndex, target = action.toCellId;
    if (!Number.isInteger(index) || index < 0 || index >= attempt.pool.length || !attempt.pool[index] || !own(attempt.board, target) || attempt.board[target] !== null || isLocked(state, target)) {
      return rejected(state, '请把待选色块放入空格');
    }
    const next = clone(state);
    snapshot(next, 'place');
    next.attempt.board[target] = next.attempt.pool[index];
    next.attempt.pool[index] = null;
    return committed(next, resolveBoard(next, true));
  }
  if (action.type === 'undo') {
    const entry = attempt.undoStack[attempt.undoStack.length - 1];
    if (!entry) return rejected(state, '还没有可以撤销的操作');
    const next = clone(state);
    next.attempt.undoStack.pop();
    next.attempt.board = { ...entry.board };
    next.attempt.pool = [...entry.pool];
    next.attempt.auxiliaryLocks = [...entry.auxiliaryLocks];
    return committed(next);
  }
  if (action.type === 'addTime') {
    const reason = propReason(state, 'addTime');
    if (reason) return rejected(state, reason);
    const next = clone(state);
    spendProp(next, 'addTime');
    next.attempt.remainingTimeMs += 30000;
    next.attempt.totalTimeGrantedMs += 30000;
    return committed(next);
  }
  if (action.type === 'peek') {
    const reason = propReason(state, 'peek');
    if (reason) return rejected(state, reason);
    if (!state.level.paths.some(p => p.id === action.pathId) || !Number.isFinite(action.nowMs)) return rejected(state, '请选择一条有效色路');
    const next = clone(state);
    spendProp(next, 'peek');
    next.attempt.status = 'PreviewingReference';
    next.attempt.preview = { pathId: action.pathId, expiresAtMs: action.nowMs + 3000 };
    return committed(next);
  }
  if (action.type === 'locate') {
    const reason = propReason(state, 'locate');
    if (reason) return rejected(state, reason);
    const candidates = state.level.cells.filter(c => !isLocked(state, c.id) && colourAt(state.level, attempt.board, c.id) !== c.targetHex);
    const target = candidates[0];
    if (!target) return rejected(state, '没有可归位的色块');
    const sourceCell = state.level.cells.find(c => !isLocked(state, c.id) && colourAt(state.level, attempt.board, c.id) === target.targetHex);
    const sourcePool = attempt.pool.findIndex(id => id !== null && state.level.tiles.find(t => t.id === id)?.hex === target.targetHex);
    if (!sourceCell && sourcePool < 0) return rejected(state, '没有可用的目标色块');
    const next = clone(state);
    snapshot(next, 'locate');
    const displacedTile = next.attempt.board[target.id];
    // Extract the correct tile first. Its old tray position becomes available before any
    // displaced tile is returned, so arbitrary tray rearrangements never require a swap.
    const correctTile = sourceCell ? next.attempt.board[sourceCell.id] : next.attempt.pool[sourcePool];
    if (sourceCell) next.attempt.board[sourceCell.id] = null;
    else next.attempt.pool[sourcePool] = null;
    if (displacedTile !== null) {
      const home = state.level.trayHomeByTile[displacedTile];
      const slot = next.attempt.pool[home] === null ? home : next.attempt.pool.findIndex(tile => tile === null);
      if (slot < 0) return rejected(state, '没有可用的待选空槽');
      next.attempt.pool[slot] = displacedTile;
    }
    next.attempt.board[target.id] = correctTile;
    next.attempt.auxiliaryLocks.push(target.id);
    spendProp(next, 'locate');
    return committed(next, resolveBoard(next, false));
  }
  return rejected(state, '未知动作');
}

function beginReward(state: GameState, action: Extract<Action, { type: 'beginReward' }>): Transition {
  const attempt = state.attempt;
  if (!validId(action.requestId) || own(state.wallet.rewardLedger, action.requestId)) return rejected(state, '奖励请求 ID 已使用或无效');
  // A detached, unconfirmed request remains busy until the platform resolves it.
  if (Object.values(state.wallet.rewardLedger).some(r => r.status === 'pending')) return rejected(state, '已有视频请求等待确认');
  if (action.rewardType === 'revive') {
    if (attempt.status !== 'Failed' || attempt.revivesUsed >= state.level.challenge.revive.maxPerAttempt) return rejected(state, '当前不能申请视频复活');
  } else if (action.rewardType === 'doubleScore') {
    if (!canClaimDoubleScore(state)) return rejected(state, '当前不能申请双倍积分');
  } else if (action.rewardType === 'restock') {
    if (attempt.status !== 'Playing' || !action.prop || !own(PROP_LIMITS, action.prop)) return rejected(state, '当前不能申请道具补货');
    if (attempt.propRestockAdsGranted >= 1) return rejected(state, '本次尝试已领取过道具补货');
    if (state.wallet.inventory[action.prop] !== 0 || attempt.propUses[action.prop] >= PROP_LIMITS[action.prop]) return rejected(state, '当前道具无需或无法补货');
  } else return rejected(state, '无效奖励类型');
  const next = clone(state);
  next.wallet.rewardLedger[action.requestId] = {
    id: action.requestId, attemptId: attempt.id, levelId: state.level.id, levelContentVersion: state.level.contentVersion,
    rewardType: action.rewardType,
    ...(action.rewardType === 'doubleScore' ? { scoreEpoch: effectiveScoreEpoch(state.level) } : {}), amount: action.rewardType === 'revive' ? state.level.challenge.revive.addSeconds : action.rewardType === 'doubleScore' ? attempt.pointsEarned : 1,
    prop: action.rewardType === 'restock' ? action.prop! : null, status: 'pending',
  };
  next.attempt.previousStatusBeforeAd = attempt.status as 'Playing' | 'Failed' | 'Won';
  next.attempt.status = 'AdPending';
  next.attempt.pendingRewardRequestId = action.requestId;
  return committed(next, [{ type: 'rewardRequested', requestId: action.requestId }]);
}
function applyRevive(state: GameState, amount: number): void {
  state.attempt.remainingTimeMs += amount * 1000;
  state.attempt.totalTimeGrantedMs += amount * 1000;
  state.attempt.revivesUsed++;
  state.attempt.status = 'Playing';
}
function settleReward(state: GameState, requestId: string, outcome: 'completed' | 'cancelled' | 'failed'): Transition {
  if (!['completed', 'cancelled', 'failed'].includes(outcome)) return rejected(state, '无效视频结算结果');
  const request = own(state.wallet.rewardLedger, requestId) ? state.wallet.rewardLedger[requestId] : undefined;
  if (!request || request.status !== 'pending') return rejected(state, '奖励请求不存在或已经处理');
  const next = clone(state);
  const sameAttempt = request.attemptId === next.attempt.id && request.levelId === next.level.id && request.levelContentVersion === next.level.contentVersion;
  if (sameAttempt && next.attempt.pendingRewardRequestId === requestId) {
    next.attempt.status = next.attempt.previousStatusBeforeAd ?? 'Failed';
    next.attempt.previousStatusBeforeAd = null;
    next.attempt.pendingRewardRequestId = null;
  }
  next.wallet.rewardLedger[requestId].status = outcome === 'completed' ? 'granted' : outcome;
  if (outcome !== 'completed') return committed(next);
  let deferred = false;
  if (request.rewardType === 'doubleScore') {
    const entry = next.wallet.levelScores[request.levelId];
    const matchingEpoch = !!entry && effectiveScoreEpoch(entry) === effectiveScoreEpoch(request);
    if (!matchingEpoch || !entry.doubleScoreClaimed) {
      // A late confirmed video from an old puzzle design keeps its earned bonus, but cannot
      // consume the new design's independent double-score qualification.
      next.wallet.points += request.amount;
      if (matchingEpoch) entry.doubleScoreClaimed = true;
    }
  } else if (request.rewardType === 'restock') {
    next.wallet.inventory[request.prop!] += request.amount;
    if (sameAttempt) next.attempt.propRestockAdsGranted++;
  } else if (sameAttempt && next.attempt.status === 'Failed' && next.attempt.revivesUsed < next.level.challenge.revive.maxPerAttempt) {
    applyRevive(next, request.amount);
    next.attempt.reviveAdsGranted++;
  } else {
    deferred = true;
    next.wallet.pendingRevives.push({ requestId, originAttemptId: request.attemptId, amount: request.amount });
  }
  return committed(next, [{ type: 'rewardGranted', requestId, deferred }]);
}

/** Progress has no engine dependency; the storage adapter decides where it is written. */
export function completeLevel(save: ProgressSave, allIds: string[]): void {
  if (!save.game || save.game.attempt.status !== 'Won') return;
  const id = save.game.level.id;
  if (!save.completed.includes(id)) save.completed.push(id);
  if (id === 'L004' && !save.starterGranted) {
    save.game.wallet.inventory.locate += 3;
    save.game.wallet.inventory.peek += 2;
    save.game.wallet.inventory.addTime += 1;
    save.starterGranted = true;
  }
  for (let chapter = 1; chapter <= 5; chapter++) {
    const ids = allIds.slice((chapter - 1) * 20, chapter * 20);
    if (ids.length === 20 && ids.every(levelId => save.completed.includes(levelId))
      && !save.chapterGifts.includes(chapter)) {
      save.game.wallet.inventory.locate++;
      save.game.wallet.inventory.peek++;
      save.game.wallet.inventory.addTime++;
      save.chapterGifts.push(chapter);
    }
  }
}

export function unlockedThrough(save: ProgressSave, allIds: string[]): number {
  const firstUnfinished = allIds.findIndex(id => !save.completed.includes(id));
  return firstUnfinished < 0 ? allIds.length - 1 : firstUnfinished;
}

import { createGame, GameState, LevelDefinition, restoreGame, restoreWallet } from '../core';
import { defaultSettings, PlayerSettings, ProgressSave, unlockedThrough } from './Progress';

function record(value: unknown, name: string): Record<string, unknown> {
  if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error(`Invalid ${name}`);
  return value as Record<string, unknown>;
}

/** Settings damage cannot erase otherwise valid earned progress or inventory. */
function restoreSettings(value: unknown): PlayerSettings {
  const defaults = defaultSettings();
  if (!value || typeof value !== 'object' || Array.isArray(value)) return defaults;
  const raw = value as Record<string, unknown>;
  return {
    sound: typeof raw.sound === 'boolean' ? raw.sound : defaults.sound,
    haptics: typeof raw.haptics === 'boolean' ? raw.haptics : defaults.haptics,
    reducedMotion: typeof raw.reducedMotion === 'boolean' ? raw.reducedMotion : defaults.reducedMotion,
  };
}

/**
 * Extracts global player rights from supported save envelopes and binds them to current content.
 * The old attempt is never interpreted using new rules. Invalid wallets throw so storage can try
 * another backup rather than silently replacing earned inventory or confirmed rewards with zero.
 */
export function restoreProgress(input: unknown, contentVersion: string,
    levels: LevelDefinition[], attemptId: string): ProgressSave {
  if (!levels.length || !contentVersion) throw new Error('Current content is required');
  const raw = record(input, 'progress');
  if (raw.version !== 1 && raw.version !== 2 && raw.version !== 3) throw new Error('Unsupported progress version');
  if (typeof raw.contentVersion !== 'string' || !raw.contentVersion) throw new Error('Invalid saved content version');
  if (typeof raw.starterGranted !== 'boolean' || !Array.isArray(raw.completed) || !Array.isArray(raw.chapterGifts)) throw new Error('Invalid global progress');
  const allIds = levels.map(level => level.id);
  const ids = new Set(allIds);
  const completed = raw.completed as unknown[];
  const chapterGifts = raw.chapterGifts as unknown[];
  if (completed.some(id => typeof id !== 'string' || !ids.has(id)) || new Set(completed).size !== completed.length) throw new Error('Invalid completed levels');
  if (chapterGifts.some(chapter => typeof chapter !== 'number' || !Number.isInteger(chapter) || chapter < 1 || chapter > 5)
      || new Set(chapterGifts).size !== chapterGifts.length) throw new Error('Invalid chapter gift markers');
  const save: ProgressSave = {
    version: 3, gameplayMode: 'click-fill', inputMode: 'tap', contentVersion,
    completed: [...completed] as string[], starterGranted: raw.starterGranted,
    chapterGifts: [...chapterGifts] as number[], settings: restoreSettings(raw.settings), game: null,
  };
  if (raw.game === null) {
    // A pristine save has no wallet yet. A played save without one cannot be reconstructed safely.
    if (save.completed.length || save.starterGranted || save.chapterGifts.length) throw new Error('Played progress is missing its wallet');
    return save;
  }
  const rawGame = record(raw.game, 'saved game');
  const unlocked = unlockedThrough(save, allIds);
  const savedLevelId = rawGame.level && typeof rawGame.level === 'object'
    ? (rawGame.level as Record<string, unknown>).id : undefined;
  const savedIndex = levels.findIndex(level => level.id === savedLevelId);
  const level = levels[savedIndex >= 0 && savedIndex <= unlocked ? savedIndex : unlocked];
  const fresh = createGame(level, attemptId);
  // Validate the wallet independently of any damaged/obsolete board or attempt metadata.
  const wallet = restoreWallet(rawGame.wallet, raw.version !== 3);
  const walletOnly = restoreGame(level, { ...fresh, wallet });
  let restored: GameState | null = null;
  if (savedLevelId === level.id) {
    try { restored = restoreGame(level, { ...rawGame, wallet }); } catch { /* Only the attempt is replaced below. */ }
  }
  if (restored) save.game = restored;
  else {
    const rawAttempt = rawGame.attempt;
    const oldAttemptId = rawAttempt && typeof rawAttempt === 'object'
      ? (rawAttempt as Record<string, unknown>).id : undefined;
    if (attemptId === oldAttemptId || Object.values(walletOnly.wallet.rewardLedger).some(request => request.attemptId === attemptId)) throw new Error('Replacement attempt ID must be fresh');
    save.game = walletOnly;
  }
  return save;
}

import { sys } from 'cc';
import { emptyProgress, ProgressSave, SAVE_KEY } from '../application/Progress';
import { LevelDefinition, recoverInterruptedGame, reduceGame } from '../core';
import { restoreProgress } from '../application/RestoreProgress';
import { SnapshotStore, StorageBackend } from './SnapshotStore';

export class LocalSave {
  error = '';
  private snapshots: SnapshotStore<ProgressSave> | null = null;

  constructor(private readonly backend: StorageBackend = sys.localStorage) {}

  load(contentVersion: string, levels: LevelDefinition[]): ProgressSave {
    this.snapshots = new SnapshotStore(this.backend, SAVE_KEY, input =>
      restoreProgress(input, contentVersion, levels, `restored-${Date.now()}-${++this.recoveryId}`));
    const latest = this.snapshots.load();
    this.error = this.snapshots.error;
    if (latest.value) return this.recoverColdStart(latest.value);
    // Earlier development snapshots contribute only validated global data, never old board rules.
    for (const key of ['shisezhijian.progress.v1', 'shisezhijian.progress.v1.backup']) {
      try {
        const value = this.backend.getItem(key);
        if (!value) continue;
        const parsed = restoreProgress(JSON.parse(value), contentVersion, levels,
          `restored-${Date.now()}-${++this.recoveryId}`);
        return this.recoverColdStart(parsed);
      } catch { this.error = '本地存档暂时不可用'; }
    }
    return emptyProgress(contentVersion);
  }

  private recoveryId = 0;

  private recoverColdStart(save: ProgressSave): ProgressSave {
    if (!save.game) return save;
    save.game = recoverInterruptedGame(save.game);
    // A cold-started process has no surviving SDK instance to resolve the old callback. Release
    // only unconfirmed requests so they cannot permanently block all future videos. Confirmed
    // grants and deferred revival entitlements remain exactly as saved; no reward is invented.
    for (const request of Object.values(save.game.wallet.rewardLedger)) {
      if (request.status === 'pending') {
        save.game = reduceGame(save.game, { type: 'settleReward', requestId: request.id, outcome: 'failed' }).state;
      }
    }
    return save;
  }

  write(save: ProgressSave): boolean {
    if (!this.snapshots) {
      this.error = '本地存档尚未读取，本次进度尚未保存';
      return false;
    }
    const saved = this.snapshots.write(save);
    this.error = this.snapshots.error;
    return saved;
  }
}

/** Synchronous storage shared by Cocos localStorage, tt storage and the test backend. */
export interface StorageBackend {

  setItem(key: string, value: string): void;

interface Snapshot<T> { key: string; revision: number; value: T; encoded: string; }
export interface SnapshotRead<T> {
  value: T | null;
  revision: number;
  warning: boolean;
  found: boolean;
}

/**
 * Two independently valid, monotonically versioned snapshots. An existing good copy is never
 * overwritten until a replacement has been written and read back byte-for-byte. The format is
 * an integrity/compatibility envelope, not an anti-cheat signature or a cloud save.
 */
export class SnapshotStore<T> {
  error = '';
  revision = 0;

  constructor(private readonly backend: StorageBackend, private readonly key: string,
    private readonly validate: (input: unknown) => T) {}

  private decode(key: string, encoded: string): Snapshot<T> {
    const raw = JSON.parse(encoded) as unknown;
    if (raw && typeof raw === 'object' && 'snapshotFormat' in raw) {
      const envelope = raw as Record<string, unknown>;
      if (envelope.snapshotFormat !== 1 || !Number.isSafeInteger(envelope.revision)
        || (envelope.revision as number) < 1 || !('payload' in envelope)) throw new Error('Invalid snapshot envelope');
      return { key, revision: envelope.revision as number, value: this.validate(envelope.payload), encoded };
    }
    // Earlier saves had a plain versioned progress payload, before the storage envelope existed.
    return { key, revision: 0, value: this.validate(raw), encoded };
  }

  private inspect(): { valid: Snapshot<T>[]; warning: boolean; found: boolean } {
    const valid: Snapshot<T>[] = [];
    let warning = false;
    let found = false;
    for (const key of [this.key, `${this.key}.backup`]) {
      try {
        const encoded = this.backend.getItem(key);
        if (encoded === null || encoded === '') continue;
        found = true;
        valid.push(this.decode(key, encoded));
      } catch { warning = true; }
    }
    valid.sort((a, b) => b.revision - a.revision || Number(a.key.endsWith('.backup')) - Number(b.key.endsWith('.backup')));
    return { valid, warning, found };
  }

  load(): SnapshotRead<T> {
    const inspected = this.inspect();
    const latest = inspected.valid[0];
    this.revision = latest?.revision ?? 0;
    this.error = inspected.warning ? '部分本地存档不可用，已尝试恢复有效副本' : '';
    return { value: latest?.value ?? null, revision: this.revision,
      warning: inspected.warning, found: inspected.found };
  }

  write(value: T): boolean {
    try {
      const inspected = this.inspect();
      const nextRevision = Math.max(this.revision, ...inspected.valid.map(snapshot => snapshot.revision)) + 1;
      if (!Number.isSafeInteger(nextRevision)) throw new Error('Snapshot revision exhausted');
      const encoded = JSON.stringify({ snapshotFormat: 1, revision: nextRevision, payload: value });
      this.decode(this.key, encoded); // Never overwrite good storage with unserialisable/invalid state.
      const latestKey = inspected.valid[0]?.key;
      // Write the older, invalid or absent slot first, preserving the newest legal previous save.
      const firstKey = latestKey === this.key ? `${this.key}.backup` : this.key;
      const secondKey = firstKey === this.key ? `${this.key}.backup` : this.key;
      let committed = 0;
      for (const key of [firstKey, secondKey]) {
        if (key === latestKey && !committed) continue;
        try {
          this.backend.setItem(key, encoded);
          const readback = this.backend.getItem(key);
          if (readback !== encoded) throw new Error('Snapshot readback mismatch');
          this.decode(key, readback);
          committed++;
        } catch {
          // Keep the newest old copy if no replacement has been verified. With empty storage,
          // trying the other slot is safe; after a verified write, mirroring is also safe.
        }
      }
      if (!committed) throw new Error('No new snapshot verified');
      this.revision = nextRevision;
      this.error = committed < 2 ? '进度已保存，备用存档暂时不可写' : '';
      return true;
    } catch {
      this.error = '本地空间不足或存储不可用，本次进度尚未保存';
      return false;
    }
  }
}

export type VideoResult = 'completed' | 'cancelled' | 'failed';
export type PresentVideoSimulation = (finish: () => void, cancel: () => void, fail: () => void) => void;

export interface RewardedVideoHandle {

  offClose?(callback: (result?: VideoCloseResult) => void): void;
  onError(callback: (error: unknown) => void): void;
  offError?(callback: (error: unknown) => void): void;

function globalRuntime(): DouyinVideoRuntime | null {
  const runtime = (globalThis as unknown as { tt?: DouyinVideoRuntime }).tt;

}

  return typeof (globalThis as unknown as { tt?: unknown }).tt !== 'undefined';

/** Full viewing is confirmed by the real SDK close result; web simulation must be explicit. */
export class RewardedVideo {
  readonly simulationAvailable: boolean;
  readonly available: boolean;
  readonly unavailableReason: string;
  private readonly runtime: DouyinVideoRuntime | null;
  private readonly adUnitId: string;
  private readonly timeoutMs: number;
  private cancelActive: (() => void) | null = null;

  constructor(options: RewardedVideoOptions = {}) {
    this.runtime = options.runtime === undefined ? globalRuntime() : options.runtime;
    this.adUnitId = options.adUnitId?.trim() ?? '';
    this.timeoutMs = Math.max(1000, options.timeoutMs ?? 180000);
    this.simulationAvailable = this.runtime === null;
    this.available = this.simulationAvailable || (this.adUnitId.length > 0 && typeof this.runtime?.createRewardedVideoAd === 'function');
    this.unavailableReason = !this.adUnitId ? '激励视频尚未配置，请免费重试' : '当前环境暂不支持激励视频，请免费重试';
  }

  get isShowing(): boolean { return this.cancelActive !== null; }

  show(onResult: (outcome: VideoResult) => void, presentSimulation?: PresentVideoSimulation): void {
    if (this.cancelActive || !this.available) { onResult('failed'); return; }
    if (this.simulationAvailable) {
      if (!presentSimulation) { onResult('failed'); return; }
      let settled = false;
      const settle = (result: VideoResult) => {
        if (settled) return;
        settled = true;
        this.cancelActive = null;
        onResult(result);
      };
      this.cancelActive = () => settle('cancelled');
      try { presentSimulation(() => settle('completed'), () => settle('cancelled'), () => settle('failed')); }
      catch { settle('failed'); }
      return;
    }
    let ad: RewardedVideoHandle;
    try { ad = this.runtime!.createRewardedVideoAd!({ adUnitId: this.adUnitId }); }
    catch { onResult('failed'); return; }
    let settled = false;
    let timer: ReturnType<typeof setTimeout> | null = null;
    const close = (result?: VideoCloseResult) => {
      // Current SDKs return count even in ordinary single-video mode; follow its documented
      // precedence. One request still grants one reward, regardless of a larger count.
      const completed = result?.count !== undefined
        ? Number.isInteger(result.count) && result.count >= 1 : result?.isEnded === true;
      settle(completed ? 'completed' : 'cancelled');
    };
    const error = (_error: unknown) => settle('failed');
    const settle = (result: VideoResult) => {
      if (settled) return;
      settled = true;
      if (timer !== null) clearTimeout(timer);
      this.cancelActive = null;
      try { ad.offClose?.(close); } catch { /* SDK cleanup must not duplicate or suppress settlement. */ }
      try { ad.offError?.(error); } catch { /* Same as above. */ }
      try { ad.destroy?.(); } catch { /* Same as above. */ }
      onResult(result);
    };
    this.cancelActive = () => settle('cancelled');
    try {
      ad.onClose(close);
      if (settled) return;
      ad.onError(error);
      if (settled) return;
      timer = setTimeout(() => settle('failed'), this.timeoutMs);
      Promise.resolve(ad.load()).then(() => {
        if (!settled) return ad.show();
        return undefined;
      }).catch(error);
    } catch (caught) { error(caught); }
  }

  /** Intentionally request only the explicitly labelled web simulation. */
  showSimulation(onResult: (outcome: VideoResult) => void, present: PresentVideoSimulation): void {
    if (!this.simulationAvailable) { onResult('failed'); return; }
    this.show(onResult, present);
  }

  dispose(): void { this.cancelActive?.(); }
}

/** Pure layout in the 390-wide design space; every tray home keeps its original page/slot. */
export function boardLayout(height: number, safeTop: number, safeBottom: number,
    columns: number, rows: number, trayCount: number) {
    const gap = 6;
    const reservedBoardTop = safeTop + 92;
    const propsY = height - safeBottom - 38;
    const actionsY = propsY - 72;
    // Footer buttons remain in a predictable location; the puzzle is a single centered group above them.
    const groupLimitBottom = actionsY - 32;
    const trayColumns = Math.min(6, Math.max(1, trayCount));
    let trayRows = Math.min(3, Math.max(1, Math.ceil(trayCount / trayColumns)));
    const trayGap = rows >= 5 ? 32 : 44;
    const maxCellSize = columns <= 4 && rows <= 2 ? 68 : 52;
    const boardSize = (visibleTrayRows: number) => {
        const trayHeight = visibleTrayRows * 44 + (visibleTrayRows - 1) * gap;
        const hasPages = trayCount > trayColumns * visibleTrayRows;
        const available = groupLimitBottom - reservedBoardTop - trayHeight - trayGap - (hasPages ? 44 : 0);
        return Math.min(maxCellSize, (342 - (columns - 1) * gap) / columns,
            (available - (rows - 1) * gap) / rows);
    };
    while (trayRows > 1 && boardSize(trayRows) < 44) trayRows--;
    const cellSize = Math.max(24, boardSize(trayRows));
    const boardWidth = columns * cellSize + (columns - 1) * gap;
    const boardHeight = rows * cellSize + (rows - 1) * gap;
    const trayHeight = trayRows * 44 + (trayRows - 1) * gap;
    const trayWidth = trayColumns * 44 + (trayColumns - 1) * gap;
    const slotsPerPage = trayColumns * trayRows;
    const hasPages = trayCount > slotsPerPage;
    const groupHeight = boardHeight + trayGap + trayHeight;
    const completeHeight = groupHeight + (hasPages ? 44 : 0);
    const groupTop = reservedBoardTop + Math.max(0,
        (groupLimitBottom - reservedBoardTop - completeHeight) / 2);
    const trayTop = groupTop + boardHeight + trayGap;
    return {
        cellSize, pitch: cellSize + gap, boardTop: groupTop, boardHeight, boardWidth,
        boardStartX: 195 - (columns - 1) * (cellSize + gap) / 2,
        boardStartY: groupTop + cellSize / 2,
        trayColumns, trayRows, trayCellSize: 44, trayPitch: 50,
        trayTop, trayHeight, trayWidth, trayGap, trayLabelY: trayTop - 18,
        slotsPerPage, propsY, actionsY, pagerY: trayTop + trayHeight + 24,
        groupTop, groupHeight, groupWidth: Math.max(boardWidth, trayWidth),
        compact: cellSize < 44,
    };
}

    const heroHeight = Math.max(64, Math.min(150, cardHeight - (hasDouble ? 458 : 400)));

    if (options.artwork) this.artwork(options.artwork, centerX, heroY, this.cardWidth - 34, heroHeight);
    else this.fallbackArt(centerX, heroY, heroHeight);

    if (this.disposed) return;

  private formatTime(value: number): string {
    const seconds = Math.max(0, Math.floor((Number.isFinite(value) ? value : 0) / 1000));
    return `${Math.floor(seconds / 60)}:${String(seconds % 60).padStart(2, '0')}`;
  }

  private artwork(frame: SpriteFrame, x: number, y: number, maxWidth: number, maxHeight: number): void {
    const node = this.element('VictoryIllustration', x, y, maxWidth, maxHeight);
    const sprite = node.addComponent(Sprite);
    sprite.sizeMode = Sprite.SizeMode.CUSTOM;
    sprite.spriteFrame = frame;
    const source = frame.originalSize;
    const scale = Math.min(maxWidth / Math.max(1, source.width), maxHeight / Math.max(1, source.height));
    node.getComponent(UITransform)!.setContentSize(source.width * scale, source.height * scale);
  }

  private fallbackArt(x: number, y: number, height: number): void {
    const scale = Math.min(1, height / 115);
    const g = this.element('CelebrationFallback', x, y, 240, height).addComponent(Graphics);
    g.fillColor = this.color('#FFF2D8'); g.ellipse(0, -7 * scale, 99 * scale, 45 * scale); g.fill();
    const colors = ['#F4B37F', '#E7B8C9', '#BDC5DE', '#A5D2BD', '#F2D690'];
    colors.forEach((hex, i) => {
      g.fillColor = this.color(hex);
      g.roundRect((i - 2) * 35 * scale - 15 * scale, -14 * scale + Math.abs(i - 2) * 8 * scale,
        30 * scale, 30 * scale, 8 * scale); g.fill();
    });
    this.spark(x - 76 * scale, y - 35 * scale, 7 * scale, '#DEAE59');
    this.spark(x + 82 * scale, y - 20 * scale, 6 * scale, '#D7B0C7');
    this.spark(x + 40 * scale, y - 46 * scale, 5 * scale, '#DEAE59');
  }

  private element(name: string, x: number, y: number, width = 1, height = 1): Node {
    const node = new Node(name); node.layer = Layers.Enum.UI_2D;
    node.addComponent(UITransform).setContentSize(width, height);
    node.setPosition(x - this.options.width / 2, this.options.height / 2 - y);
    this.node.addChild(node); return node;
  }
  private color(value: string | Color): Color { return typeof value === 'string' ? new Color().fromHEX(value) : value; }
  private round(name: string, x: number, y: number, width: number, height: number,
    radius: number, fill: string | Color): Node {
    const node = this.element(name, x, y, width, height);
    const g = node.addComponent(Graphics); g.fillColor = this.color(fill);
    g.roundRect(-width / 2, -height / 2, width, height, radius); g.fill(); return node;
  }
  private label(value: string, x: number, y: number, width: number, height: number,
    fontSize: number, hex: string, bold = false): void {
    this.options.onText?.(value);
    const label = this.element('VictoryText', x, y, width, height).addComponent(Label);
    label.string = value; label.useSystemFont = true; label.fontFamily = 'sans-serif';
    label.fontSize = fontSize; label.lineHeight = fontSize * 1.2; label.isBold = bold;
    label.color = this.color(hex); label.horizontalAlign = Label.HorizontalAlign.CENTER;
    label.verticalAlign = Label.VerticalAlign.CENTER; label.enableWrapText = false;
    label.overflow = Label.Overflow.SHRINK;
  }
  private hit(label: string, x: number, y: number, width: number, height: number, callback: () => void): void {
    const node = this.element(label, x, y, width, height);
    const invoke = () => { if (!this.disposed && this.node.isValid) callback(); };
    node.on(Node.EventType.TOUCH_END, (event: EventTouch) => { event.propagationStopped = true; invoke(); });
    this.options.registerControl?.(label, invoke, { x, y, width, height });
  }
  private button(label: string, x: number, y: number, width: number, height: number,
    fill: string, callback: () => void): void {
    this.round(label, x, y, width, height, 19, fill);
    this.hit(label, x, y, width, height, callback);
  }
  private coin(x: number, y: number, radius: number): void {
    const g = this.element('PointsCoin', x, y).addComponent(Graphics);
    g.fillColor = this.color('#E8CA80'); g.circle(0, 0, radius); g.fill();
    g.strokeColor = this.color('#FFF5D4'); g.lineWidth = 1.5; g.circle(0, 0, radius - 3); g.stroke();
    g.moveTo(0, -4); g.lineTo(0, 4); g.stroke();
  }
  private play(x: number, y: number, hex: string): void {
    const g = this.element('RewardVideoPlay', x, y).addComponent(Graphics); g.fillColor = this.color(hex);
    g.moveTo(-5, -7); g.lineTo(7, 0); g.lineTo(-5, 7); g.close(); g.fill();
  }
  private chevron(x: number, y: number, hex: string): void {
    const g = this.element('NextChevron', x, y).addComponent(Graphics); g.strokeColor = this.color(hex); g.lineWidth = 2.3;
    g.moveTo(-3, -5); g.lineTo(3, 0); g.lineTo(-3, 5); g.stroke();
  }
  private spark(x: number, y: number, radius: number, hex: string): void {
    const g = this.element('VictorySpark', x, y).addComponent(Graphics); g.fillColor = this.color(hex);
    g.moveTo(0, radius); g.lineTo(radius / 3, radius / 3); g.lineTo(radius, 0);
    g.lineTo(radius / 3, -radius / 3); g.lineTo(0, -radius); g.lineTo(-radius / 3, -radius / 3);
    g.lineTo(-radius, 0); g.lineTo(-radius / 3, radius / 3); g.close(); g.fill();
  }
